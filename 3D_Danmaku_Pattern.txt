#TouhouDanmakufu[Single]
#Title["月符「月の波紋」"]
#Text["自機狙い散弾と全方位回転リングの複合"]

// Phase 1 (HP 3000-2000): 自機狙い3-Way + 12-Way回転リング
// Phase 2 (HP 2000-1000): 自機狙い5-Way + 二重リング
// Phase 3 (HP 1000-  0 ): 自機狙い7-Way + 三重高速リング

task main {
    let boss = ObjEnemy_Create();
    ObjEnemy_SetPosition(boss, 192, 120);
    ObjEnemy_SetLife(boss, 3000);
    ObjEnemy_SetCollisionA(boss, 22);
    ObjEnemy_SetTimer(boss, 45);

    BossFloat(boss);
    MainPattern(boss);

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        yield;
    }
    Obj_Delete(boss);
}

task BossFloat(boss) {
    let t = 0;
    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        ObjEnemy_SetPosition(boss,
            192 + 70 * sin(t * 1.2),
            120 + 20 * sin(t * 2.5 + 90));
        t++;
        yield;
    }
}

task MainPattern(boss) {
    let timer    = 0;
    let ringAngle = 0;

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let bx   = ObjEnemy_GetX(boss);
        let by   = ObjEnemy_GetY(boss);
        let life = ObjEnemy_GetInfo(boss, INFO_LIFE);

        if(life > 2000) {
            if(timer % 90 == 0) {
                AimedNWay(bx, by, 3, 20, 2.5, 1);
            }
            if(timer % 48 == 0) {
                Ring(bx, by, 12, ringAngle, 2.0, 3);
                ringAngle += 15;
            }
        }
        else if(life > 1000) {
            if(timer % 70 == 0) {
                AimedNWay(bx, by, 5, 18, 3.0, 1);
            }
            if(timer % 36 == 0) {
                Ring(bx, by, 16, ringAngle,         2.0, 2);
                Ring(bx, by, 16, ringAngle + 11.25, 2.8, 3);
                ringAngle += 11.25;
            }
        }
        else {
            if(timer % 50 == 0) {
                AimedNWay(bx, by, 7, 15, 3.5, 1);
            }
            if(timer % 24 == 0) {
                Ring(bx, by, 20, ringAngle,      2.0, 2);
                Ring(bx, by, 20, ringAngle + 9,  3.0, 3);
                Ring(bx, by, 20, ringAngle + 18, 3.8, 4);
                ringAngle += 9;
            }
        }

        timer++;
        yield;
    }
}

function AimedNWay(x, y, n, spread, speed, graphic) {
    let base = atan2(GetPlayerY() - y, GetPlayerX() - x);
    loop(n) {
        let offset = (loop_index - (n - 1) / 2.0) * spread;
        CreateShotA(x, y, speed, base + offset, graphic, 4);
    }
}

function Ring(x, y, n, offset, speed, graphic) {
    loop(n) {
        CreateShotA(x, y, speed, (loop_index / n) * 360 + offset, graphic, 0);
    }
}
