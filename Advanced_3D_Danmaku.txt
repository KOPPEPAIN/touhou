#TouhouDanmakufu[Single]
#Title["境符「波と粒子の境界」"]
#Text["画面を横断するsin波ストリームと自機狙いの複合"]

// Phase 1 (HP 3500-2300): 左右2ストリーム波状弾 + 自機狙い3-Way
// Phase 2 (HP 2300-1200): 左右4ストリーム波状弾 + 自機狙い5-Way
// Phase 3 (HP 1200-  0 ): 左右6ストリーム高速波 + 自機狙い7-Way + 追加リング
//
// 波状弾の仕組み:
//   毎フレーム画面端から1発ずつ発射し、発射Y座標をsin波で上下させる。
//   複数ストリームの位相をずらすことで美しい干渉縞が生まれる。

task main {
    let boss = ObjEnemy_Create();
    ObjEnemy_SetPosition(boss, 192, 100);
    ObjEnemy_SetLife(boss, 3500);
    ObjEnemy_SetCollisionA(boss, 22);
    ObjEnemy_SetTimer(boss, 60);

    BossSway(boss);
    WaveStream(boss);
    AimedBreak(boss);

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        yield;
    }
    Obj_Delete(boss);
}

task BossSway(boss) {
    let t = 0;
    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        ObjEnemy_SetPosition(boss,
            192 + 100 * sin(t * 0.6),
            100 +  10 * sin(t * 1.0));
        t++;
        yield;
    }
}

task WaveStream(boss) {
    let timer = 0;

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let life = ObjEnemy_GetInfo(boss, INFO_LIFE);

        let numStreams = 2;
        let spd       = 3.0;
        let freq      = 2.5;
        if(life <= 2300) { numStreams = 4; spd = 3.5; }
        if(life <= 1200) { numStreams = 6; spd = 4.2; freq = 3.0; }

        loop(numStreams) {
            let phase = (loop_index / numStreams) * 360;
            let y     = 224 + 130 * sin(timer * freq + phase);

            if(loop_index % 2 == 0) {
                CreateShotA(  0, y, spd,   0, 2, 0);
            }
            else {
                CreateShotA(384, y, spd, 180, 3, 0);
            }
        }

        if(life <= 1200 && timer % 30 == 0) {
            let bx = ObjEnemy_GetX(boss);
            let by = ObjEnemy_GetY(boss);
            Ring(bx, by, 12, timer * 10, 2.0, 4);
        }

        timer++;
        yield;
    }
}

task AimedBreak(boss) {
    let timer = 0;

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let bx   = ObjEnemy_GetX(boss);
        let by   = ObjEnemy_GetY(boss);
        let life = ObjEnemy_GetInfo(boss, INFO_LIFE);

        let n        = 3;
        let spread   = 22;
        let spd      = 2.5;
        let interval = 90;
        if(life <= 2300) { n = 5; spread = 18; spd = 3.0; interval = 70; }
        if(life <= 1200) { n = 7; spread = 14; spd = 3.8; interval = 50; }

        if(timer % interval == 0) {
            AimedNWay(bx, by, n, spread, spd, 1);
        }

        timer++;
        yield;
    }
}

function AimedNWay(x, y, n, spread, speed, graphic) {
    let base = atan2(GetPlayerY() - y, GetPlayerX() - x);
    loop(n) {
        let offset = (loop_index - (n - 1) / 2.0) * spread;
        CreateShotA(x, y, speed, base + offset, graphic, 4);
    }
}

function Ring(x, y, n, offset, speed, graphic) {
    loop(n) {
        CreateShotA(x, y, speed, (loop_index / n) * 360 + offset, graphic, 0);
    }
}
