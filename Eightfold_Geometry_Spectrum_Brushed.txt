function wait(frames) {
    let i = 0;
    while (i < frames) {
        yield;
        i++;
    }
}

function get_3d_scale(z) {
    let focal = 450.0;
    if (z <= -focal) {
        return 0.01;
    }
    let scale = focal / (focal + z);
    if (scale > 2.4) {
        return 2.4;
    }
    if (scale < 0.32) {
        return 0.32;
    }
    return scale;
}

function is_safe_depth(z) {
    return z > -255;
}

function rotate_3d(x, y, z, ax, ay, az) {
    let sx = sin(ax); let cx = cos(ax);
    let sy = sin(ay); let cy = cos(ay);
    let sz = sin(az); let cz = cos(az);

    let y1 = y * cx - z * sx;
    let z1 = y * sx + z * cx;
    let x2 = x * cy + z1 * sy;
    let z2 = -x * sy + z1 * cy;
    let x3 = x2 * cz - y1 * sz;
    let y3 = x2 * sz + y1 * cz;

    return [x3, y3, z2];
}

task TAimedShot(x, y, speed, angle, color, life) {
    let shot = CreateShotA1(x, y, speed, angle, color, 24);
    ObjRender_SetBlendType(shot, BLEND_ADD_ARGB);
    let t = 0;
    while (!Obj_IsDeleted(shot) && t < life) {
        let depth = sin(t * 4);
        ObjRender_SetAlpha(shot, 190 + depth * 60);
        ObjRender_SetScaleXYZ(shot, 1.0 + depth * 0.15, 1.0 + depth * 0.15, 1);
        t++;
        yield;
    }
}

task TAimedFan(cx, cy, color, count, spread, speed, life) {
    let baseAngle = GetAngleToPlayer(objBoss);
    let i = 0;
    while (i < count) {
        let offset = 0;
        if (count > 1) {
            offset = (i - (count - 1) / 2.0) * spread;
        }
        TAimedShot(cx, cy, speed, baseAngle + offset, color, life);
        i++;
    }
}

task TAimedRing(cx, cy, color, count, speed, life, rotOffset) {
    let i = 0;
    while (i < count) {
        let angle = i * 360 / count + rotOffset;
        TAimedShot(cx, cy, speed, angle, color, life);
        i++;
    }
}

    while (!Obj_IsDeleted(shot) && t < life) {

        ObjRender_SetScaleXYZ(shot, 1 + depth * 0.3, 1 + depth * 0.3, 1);

}

task TWaveShot(x, y, vx, vy, color, freq, amp, life) {
    let shot = CreateShotA1(x, y, 0, 0, color, 24);
    ObjRender_SetBlendType(shot, BLEND_ADD_ARGB);
    let t = 0;
    let baseX = x;
    let baseY = y;
    let norm = ((vx * vx + vy * vy) ^ 0.5);
    let nx = 0;
    let ny = 0;
    if (norm > 0) {
        nx = -vy / norm;
        ny = vx / norm;
    }
    while (!Obj_IsDeleted(shot) && t < life) {
        baseX += vx;
        baseY += vy;
        let offset = sin(t * freq) * amp;
        ObjMove_SetPosition(shot, baseX + nx * offset, baseY + ny * offset);
        let depth = cos(t * 4);
        ObjRender_SetAlpha(shot, 140 + depth * 110);
        ObjRender_SetScaleXYZ(shot, 1 + depth * 0.4, 1 + depth * 0.4, 1);
        t++;
        yield;
    }
}

    ObjText_SetFontBold(objText, true);
    ObjText_SetFontColorTop(objText, 255, 255, 255);
    ObjText_SetFontColorBottom(objText, 210, 230, 255);
    ObjText_SetFontBorderWidth(objText, 2);
    ObjText_SetFontBorderColor(objText, 18, 16, 28);
    ObjText_SetHorizontalAlignment(objText, ALIGNMENT_RIGHT);
    ObjText_SetMaxWidth(objText, 370);
    ObjRender_SetX(objText, 10);
    ObjRender_SetY(objText, 20);

    let alpha = 0;
    while (alpha < 255 && !Obj_IsDeleted(objBoss)) {
        alpha += 17;
        ObjRender_SetAlpha(objText, alpha);
        yield;
    }
    wait(112);

        alpha -= 11;
        ObjRender_SetAlpha(objText, alpha);
        yield;
    }
    Obj_Delete(objText);
}

function get_flame_star() {
    let v = []; let e = []; let n = 6;
    let i = 0;
    while (i < n) {
        let a = i * 360 / n;
        v = v ~ [[cos(a) * 1.6, sin(a) * 1.6, 0.3 * sin(a * 2)]];
        i++;
    }
    i = 0;
    while (i < n) {
        let a = i * 360 / n + 180 / n;
        v = v ~ [[cos(a) * 0.6, sin(a) * 0.6, -0.3 * sin(a * 2)]];
        i++;
    }
    v = v ~ [[0, 0, 0]];
    let center = 12;
    i = 0;
    while (i < n) {
        let next = (i + 1) % n;
        e = e ~ [[i, center]];
        e = e ~ [[n + i, center]];
        e = e ~ [[i, n + i]];
        e = e ~ [[n + i, next]];
        i++;
    }
    return [v, e];
}

function get_vortex_spiral() {
    let v = []; let e = []; let n = 16; let turns = 1.5;
    let i = 0;
    while (i < n) {
        let a = i * 360 * turns / (n - 1);
        let r = 0.3 + 1.2 * i / (n - 1);
        let z = -1 + 2 * i / (n - 1);
        v = v ~ [[cos(a) * r, sin(a) * r, z]];
        i++;
    }
    i = 0;
    while (i < n - 1) {
        e = e ~ [[i, i + 1]];
        i++;
    }
    v = v ~ [[0, 0, -1.2]];
    v = v ~ [[0, 0, 1.2]];
    e = e ~ [[16, 0]];
    e = e ~ [[15, 17]];
    return [v, e];
}

function get_lightning_lattice() {
    let path = [[-1,1.6,0],[1,1.0,0.3],[-1,0.4,-0.3],[1,-0.2,0.3],[-1,-0.8,-0.3],[1,-1.4,0.3],[-1,-2.0,0]];
    let v = []; let e = []; let n = length(path);
    let i = 0;
    while (i < n) {
        v = v ~ [[path[i][0] * 0.7, path[i][1] * 0.7, path[i][2] * 0.7]];
        i++;
    }
    i = 0;
    while (i < n) {
        v = v ~ [[path[i][0] * 0.7 + 0.5, path[i][1] * 0.7, path[i][2] * 0.7 - 0.6]];
        i++;
    }
    i = 0;
    while (i < n - 1) {
        e = e ~ [[i, i + 1]];
        e = e ~ [[n + i, n + i + 1]];
        i++;
    }
    i = 0;
    while (i < n) {
        e = e ~ [[i, n + i]];
        i++;
    }
    return [v, e];
}

function get_vine_helix() {
    let v = []; let e = []; let n = 12; let turns = 1.2;
    let i = 0;
    while (i < n) {
        let a = i * 360 * turns / (n - 1);
        let z = -1 + 2 * i / (n - 1);
        v = v ~ [[cos(a) * 0.8, sin(a) * 0.8, z]];
        i++;
    }
    i = 0;
    while (i < n - 1) {
        e = e ~ [[i, i + 1]];
        i++;
    }
    i = 0;
    while (i < n) {
        if (i % 2 == 0) {
            let a = i * 360 * turns / (n - 1);
            let z = -1 + 2 * i / (n - 1);
            v = v ~ [[cos(a) * 1.6, sin(a) * 1.6, z]];
            e = e ~ [[i, n + length(v) - n - 1]];
        }
        i++;
    }
    return [v, e];
}

function get_cloud_ring() {
    let v = []; let e = []; let div = 8; let r = 0.55;
    let centers = [[0,0,0],[1.0,0.4,0.3],[-1.0,0.3,-0.3],[0.3,-0.9,0.2]];
    let c = 0;
    while (c < length(centers)) {
        let base = length(v);
        let i = 0;
        while (i < div) {
            let a = i * 360 / div;
            v = v ~ [[centers[c][0] + cos(a) * r, centers[c][1] + sin(a) * r, centers[c][2]]];
            i++;
        }
        i = 0;
        while (i < div) {
            let next = (i + 1) % div;
            e = e ~ [[base + i, base + next]];
            i++;
        }
        c++;
    }
    return [v, e];
}

function get_ice_crystal() {
    let v = [[0, 0, 0]]; let e = []; let arms = 6;
    let i = 0;
    while (i < arms) {
        let a = i * 360 / arms;
        let tip = length(v);
        v = v ~ [[cos(a) * 1.6, sin(a) * 1.6, 0]];
        e = e ~ [[0, tip]];
        let mid = length(v);
        v = v ~ [[cos(a) * 0.9, sin(a) * 0.9, 0]];
        e = e ~ [[0, mid]];
        let a1 = a + 35;
        v = v ~ [[cos(a1) * 0.9, sin(a1) * 0.9, 0.4]];
        e = e ~ [[mid, length(v) - 1]];
        let a2 = a - 35;
        v = v ~ [[cos(a2) * 0.9, sin(a2) * 0.9, -0.4]];
        e = e ~ [[mid, length(v) - 1]];
        i++;
    }
    return [v, e];
}

function get_galaxy_spiral() {
    let v = [[0, 0, 0]]; let e = []; let arm = 0;
    while (arm < 2) {
        let n = 10; let i = 0;
        let first = -1;
        while (i < n) {
            let t = i / (n - 1);
            let a = t * 300 + arm * 180;
            let r = 0.3 + 1.3 * t;
            let z = t * 0.6;
            if (arm == 1) { z = -z; }
            v = v ~ [[cos(a) * r, sin(a) * r, z]];
            let idx = length(v) - 1;
            if (first == -1) {
                first = idx;
                e = e ~ [[0, idx]];
            } else {
                e = e ~ [[idx - 1, idx]];
            }
            i++;
        }
        arm++;
    }
    return [v, e];
}

function get_halo_rings() {
    let v = []; let e = []; let div = 12;
    let radii = [0.6, 1.0, 1.4];
    let tilts = [0, 30, -30];
    let r = 0;
    while (r < 3) {
        let base = length(v);
        let i = 0;
        while (i < div) {
            let a = i * 360 / div;
            v = v ~ [[cos(a) * radii[r], sin(a) * radii[r] * cos(tilts[r]), sin(a) * radii[r] * sin(tilts[r])]];
            i++;
        }
        i = 0;
        while (i < div) {
            let next = (i + 1) % div;
            e = e ~ [[base + i, base + next]];
            i++;
        }
        r++;
    }
    return [v, e];
}

function GetPhaseShape(phase) {
    if (phase == 1) { return get_flame_star(); }
    if (phase == 2) { return get_vortex_spiral(); }
    if (phase == 3) { return get_lightning_lattice(); }
    if (phase == 4) { return get_vine_helix(); }
    if (phase == 5) { return get_cloud_ring(); }
    if (phase == 6) { return get_ice_crystal(); }
    if (phase == 7) { return get_galaxy_spiral(); }
    return get_halo_rings();
}

function CreateHiddenShot(color) {
    let shot = CreateShotA1(-999, -999, 0, 0, color, 0);
    ObjShot_SetIntersectionEnable(shot, false);
    ObjShot_SetAutoDelete(shot, false);
    ObjRender_SetBlendType(shot, BLEND_ADD_ARGB);
    ObjRender_SetAlpha(shot, 0);
    return shot;
}

function CreateHiddenLaser(graphicID) {
    let laser = ObjShot_Create(OBJ_STRAIGHT_LASER);
    ObjShot_SetGraphic(laser, graphicID);
    ObjShot_SetAutoDelete(laser, false);
    ObjShot_SetIntersectionEnable(laser, false);
    ObjStLaser_SetSource(laser, false);
    ObjRender_SetBlendType(laser, BLEND_ADD_ARGB);
    ObjRender_SetAlpha(laser, 0);
    ObjShot_Regist(laser);
    return laser;
}

function DrawLaserEdge(obj, x1, y1, x2, y2, z, width, activeCollision) {
    let dx = x2 - x1;
    let dy = y2 - y1;
    let len = (dx * dx + dy * dy) ^ 0.5;

    if (!is_safe_depth(z) || len < 1.5 || len > 420) {
        ObjRender_SetAlpha(obj, 0);
        ObjShot_SetIntersectionEnable(obj, false);
        return;
    }

    let angle = atan2(dy, dx);
    ObjMove_SetPosition(obj, x1, y1);
    ObjStLaser_SetAngle(obj, angle);
    ObjLaser_SetLength(obj, len);
    ObjLaser_SetRenderWidth(obj, width);

    if (z < 0) {
        ObjRender_SetAlpha(obj, 235);
        ObjShot_SetIntersectionEnable(obj, activeCollision);
    } else {
        ObjRender_SetAlpha(obj, 50);
        ObjShot_SetIntersectionEnable(obj, false);
    }
}

task TBulletWireframeStatic(vBase, eBase, color, phase, baseR) {
    let laserColor = GetPhaseLaserColor(phase);
    let edgeLasers = [];
    let vertexShots = [];
    let numEdges = length(eBase);
    let i = 0;

    while (i < numEdges) {
        let laser = CreateHiddenLaser(laserColor);
        edgeLasers = edgeLasers ~ [laser];
        i++;
    }

    i = 0;
    while (i < length(vBase)) {
        let vertexShot = CreateHiddenShot(color);
        vertexShots = vertexShots ~ [vertexShot];
        i++;
    }

    let localFrame = 0;
    while (globalPhaseID == phase && !Obj_IsDeleted(objBoss)) {
        let cx = ObjMove_GetX(objBoss);
        let cy = ObjMove_GetY(objBoss);
        let pulse = 0.5 + 0.5 * sin(localFrame * 3);

        i = 0;
        while (i < length(vBase)) {
            let vertex = vBase[i];
            let p = rotate_3d(vertex[0] * baseR, vertex[1] * baseR, vertex[2] * baseR, globalAx, globalAy, globalAz);
            let scale = get_3d_scale(p[2]);
            let shot = vertexShots[i];
            ObjMove_SetPosition(shot, cx + p[0] * scale, cy + p[1] * scale);
            ObjRender_SetScaleXYZ(shot, 1.35 + pulse * 0.18, 1.35 + pulse * 0.18, 1);
            if (p[2] > 0) {
                ObjRender_SetAlpha(shot, 60 + pulse * 20);
            } else {
                ObjRender_SetAlpha(shot, 190 + pulse * 40);
            }
            i++;
        }

        i = 0;
        while (i < numEdges) {
            let v1 = vBase[eBase[i][0]];
            let v2 = vBase[eBase[i][1]];
            let p1 = rotate_3d(v1[0] * baseR, v1[1] * baseR, v1[2] * baseR, globalAx, globalAy, globalAz);
            let p2 = rotate_3d(v2[0] * baseR, v2[1] * baseR, v2[2] * baseR, globalAx, globalAy, globalAz);
            let scale1 = get_3d_scale(p1[2]);
            let scale2 = get_3d_scale(p2[2]);
            let sx1 = cx + p1[0] * scale1;
            let sy1 = cy + p1[1] * scale1;
            let sx2 = cx + p2[0] * scale2;
            let sy2 = cy + p2[1] * scale2;
            let edgeZ = (p1[2] + p2[2]) * 0.5;
            DrawLaserEdge(edgeLasers[i], sx1, sy1, sx2, sy2, edgeZ, 7.5 + pulse * 2.2, false);
            i++;
        }
        localFrame++;
        yield;
    }

    i = 0;
    while (i < length(edgeLasers)) {
        Obj_Delete(edgeLasers[i]);
        i++;
    }

    i = 0;
    while (i < length(vertexShots)) {
        Obj_Delete(vertexShots[i]);
        i++;
    }
}

task TGeometryWave(vBase, eBase, color, phase, baseR, moveType) {
    let laserColor = GetPhaseLaserColor(phase);
    let edgeLasers = [];
    let vertexShots = [];
    let numEdges = length(eBase);
    let i = 0;

    while (i < numEdges) {
        let laser = CreateHiddenLaser(laserColor);
        edgeLasers = edgeLasers ~ [laser];
        i++;
    }

    i = 0;
    while (i < length(vBase)) {
        let vertexShot = CreateHiddenShot(color);
        vertexShots = vertexShots ~ [vertexShot];
        i++;
    }

    let originX = ObjMove_GetX(objBoss);
    let originY = ObjMove_GetY(objBoss);

    let ax = rand(0, 360);
    let ay = rand(0, 360);
    let az = rand(0, 360);

    let destAngle = 0;
    if (moveType == 0) {
        destAngle = GetAngleToPlayer(objBoss);
    } else {
        destAngle = rand(0, 360);
    }
    let destR = rand(70, 130);
    let destX = destR * cos(destAngle);
    let destY = destR * sin(destAngle);

    let life = 130;
    let t = 0;
    while (globalPhaseID == phase && t < life && !Obj_IsDeleted(objBoss)) {
        let progress = t / life;
        let bx = destX * progress;
        let by = destY * progress;
        ax += 3.2;
        ay += 2.4;
        az += 1.8;
        let pulse = 0.6 + 0.4 * sin(t * 5);

        i = 0;
        while (i < length(vBase)) {
            let vertex = vBase[i];
            let p = rotate_3d(vertex[0] * baseR, vertex[1] * baseR, vertex[2] * baseR, ax, ay, az);
            let shot = vertexShots[i];
            ObjMove_SetPosition(shot, originX + p[0] + bx, originY + p[1] + by);
            ObjRender_SetScaleXYZ(shot, 1.6, 1.6, 1);
            ObjRender_SetAlpha(shot, 160 + pulse * 70);
            i++;
        }

        i = 0;
        while (i < numEdges) {
            let v1 = vBase[eBase[i][0]];
            let v2 = vBase[eBase[i][1]];
            let p1 = rotate_3d(v1[0] * baseR, v1[1] * baseR, v1[2] * baseR, ax, ay, az);
            let p2 = rotate_3d(v2[0] * baseR, v2[1] * baseR, v2[2] * baseR, ax, ay, az);
            let sx1 = originX + p1[0] + bx;
            let sy1 = originY + p1[1] + by;
            let sx2 = originX + p2[0] + bx;
            let sy2 = originY + p2[1] + by;
            DrawLaserEdge(edgeLasers[i], sx1, sy1, sx2, sy2, -10, 7.0 + pulse * 1.5, true);
            i++;
        }
        t++;
        yield;
    }

    i = 0;
    while (i < length(edgeLasers)) {
        Obj_Delete(edgeLasers[i]);
        i++;
    }

    i = 0;
    while (i < length(vertexShots)) {
        Obj_Delete(vertexShots[i]);
        i++;
    }
}

task TPattern_Generic(phase) {
    let shape = GetPhaseShape(phase);
    let color = GetPhaseColor(phase);
    let accent = GetPhaseAccentColor(phase);
    let radius = GetPhaseRadius(phase);
    let waveRadius = GetWaveRadius(phase);

    TBulletWireframeStatic(shape[0], shape[1], color, phase, radius);

    let t = 0;
    while (globalPhaseID == phase && !Obj_IsDeleted(objBoss)) {
        globalAx += 1.0 + phase * 0.12 + 0.35 * sin(t * 2);
        globalAy += 1.45 + phase * 0.10;
        globalAz += 0.85 + phase * 0.08 + 0.2 * cos(t * 3);

        if (t > 30 && t % GetWaveInterval(phase) == 0) {
            TGeometryWave(shape[0], shape[1], color, phase, waveRadius, 0);
            TGeometryWave(shape[0], shape[1], accent, phase, waveRadius, 1);
        }

        t++;
        yield;
    }
}

task TBossWander() {
    let t = 0;
    while (!Obj_IsDeleted(objBoss)) {
        if (globalPhaseID > 0) {
            let driftX = BOSS_X + cos(t * 67) * rand(34, 84);
            let driftY = BOSS_Y + sin(t * 83) * rand(18, 54);
            if (globalPhaseID == 8) {
                driftX = BOSS_X;
                driftY = BOSS_Y;
            }
            ObjMove_SetDestAtFrame(objBoss, driftX, driftY, 92);
            t++;
        }
        wait(150);
    }
}

task TMainSequence() {
    ObjMove_SetDestAtFrame(objBoss, BOSS_X, BOSS_Y, 60);
    wait(60);
    TBossWander();

    while (!Obj_IsDeleted(objBoss)) {
        let phase = 1;
        while (phase <= 8 && !Obj_IsDeleted(objBoss)) {
            ExecutePhase(phase);
            phase++;
        }
    }
}

function ExecutePhase(phase) {
    if (Obj_IsDeleted(objBoss)) {
        return;
    }

    globalPhaseID = phase;
    globalAx = phase * 17;
    globalAy = phase * 29;
    globalAz = phase * 41;
    TAnnounce(GetPhaseName(phase), GetPhaseColor(phase));
    TPattern_Generic(phase);

    wait(PHASE_FRAMES);
    if (Obj_IsDeleted(objBoss)) {
        return;
    }

    globalPhaseID = 0;
    ClearCurrentPhase(phase, GetPhaseColor(phase));
}

function SetupBossSprite() {
    ObjPrim_SetTexture(objBoss, "script/default_system/img/Default_Shot.png");
    ObjSprite2D_SetSourceRect(objBoss, 0, 0, 12, 12);
    ObjSprite2D_SetDestCenter(objBoss);
    ObjRender_SetScaleXYZ(objBoss, 3.0, 3.0, 1);
    ObjRender_SetColor(objBoss, 240, 230, 255);
}

@Initialize {
    SetPlayerLife(8);
    objScene = GetEnemyBossSceneObjectID();
    objBoss = ObjEnemy_Create(OBJ_ENEMY_BOSS);
    ObjEnemy_Regist(objBoss);
    ObjMove_SetPosition(objBoss, BOSS_X, -36);
    ObjMove_SetDestAtFrame(objBoss, BOSS_X, BOSS_Y, 60);
    ObjEnemyBossScene_Add(objScene, 0, objBoss);
    ObjEnemy_SetDamageRate(objBoss, 0, 0);
    LoadEnemyShotData(GetModuleDirectory() ~ "script/default_system/Default_ShotData.txt");
    SetupBossSprite();
    TMainSequence();
}

@Event {
    let event = GetEventType();
    if (event == EV_REQUEST_LIFE) {
        SetScriptResult(10000);
    } else if (event == EV_REQUEST_TIMER) {
        SetScriptResult(999);
    } else if (event == EV_REQUEST_SPELL_SCORE) {
        SetScriptResult(30000000);
    }
}

@MainLoop {
    let bossX = ObjMove_GetX(objBoss);
    let bossY = ObjMove_GetY(objBoss);
    let scale = 3.0 + 0.22 * sin(frameCounter * 5);

    ObjEnemy_SetIntersectionCircleToShot(objBoss, bossX, bossY, 32);
    ObjEnemy_SetIntersectionCircleToPlayer(objBoss, bossX, bossY, 24);
    ObjRender_SetScaleXYZ(objBoss, scale, scale, 1);
    ObjRender_SetAngleZ(objBoss, frameCounter * 2.5);

    frameCounter++;
    yield;
}

@Finalize {
    if (!Obj_IsDeleted(objBoss)) {
        Obj_Delete(objBoss);
    }
}
