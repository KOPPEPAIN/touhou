#TouhouDanmakufu[Single]
#Title["桜符「散る花びら」"]
#Text["幽々子風の花びら弾幕\n大型弾の合間を縫う散弾と自機狙い"]

// ===== スペルカード構成 =====
// Phase 1 (HP 3000-2000): ゆっくりした大型弾（花びら）+ ランダム散弾
// Phase 2 (HP 2000-1000): 大型弾 + 中型自機狙い散弾
// Phase 3 (HP 1000-0  ): 全パターン高速化 + 自機狙い追加

let BOSS_START_X = 192;
let BOSS_START_Y = 110;

task main {
    let boss = ObjEnemy_Create();
    ObjEnemy_SetPosition(boss, BOSS_START_X, BOSS_START_Y);
    ObjEnemy_SetLife(boss, 3000);
    ObjEnemy_SetCollisionA(boss, 22);
    ObjEnemy_SetTimer(boss, 50);

    BossGlide(boss);
    PetalPattern(boss);
    ScatterPattern(boss);
    AimedPattern(boss);

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        yield;
    }

    Obj_Delete(boss);
}

// ゆっくりした横移動（幽々子らしい優雅な動き）
task BossGlide(boss) {
    let t = 0;
    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let x = BOSS_START_X + 90 * sin(t * 0.8);
        let y = BOSS_START_Y + 15 * sin(t * 1.5);
        ObjEnemy_SetPosition(boss, x, y);
        t++;
        yield;
    }
}

// 大型弾「花びら」パターン - 全方向にゆっくり大型弾を発射
task PetalPattern(boss) {
    let timer = 0;
    let angle = 0;

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let bx = ObjEnemy_GetX(boss);
        let by = ObjEnemy_GetY(boss);
        let life = ObjEnemy_GetInfo(boss, INFO_LIFE);

        let interval = life > 1000 ? 20 : 14; // Phase 3で速くなる
        let speed    = life > 1000 ? 1.2 : 1.8;

        if(timer % interval == 0) {
            // 8方向に大型弾 + 45度オフセットで8方向 = 16方向の花びら
            loop(8) {
                CreateShotA(bx, by, speed, angle + loop_index * 45,     5, 6);
                CreateShotA(bx, by, speed, angle + loop_index * 45 + 22.5, 5, 6);
            }
            angle += 5; // ゆっくり回転させる
        }

        timer++;
        yield;
    }
}

// ランダム散弾 - 自機周囲に"降り注ぐ"ような弾幕
task ScatterPattern(boss) {
    let timer = 0;

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let bx = ObjEnemy_GetX(boss);
        let by = ObjEnemy_GetY(boss);
        let life = ObjEnemy_GetInfo(boss, INFO_LIFE);

        // Phase 1のみランダム散弾
        if(life > 2000 && timer % 8 == 0) {
            // 真下向き ± ランダム で"花びらが降る"演出
            let numShots = 5;
            loop(numShots) {
                let angle = 90 + rand(-35, 35);
                let speed = rand(1.5, 2.5);
                CreateShotA(bx, by, speed, angle, 4, 0);
            }
        }

        timer++;
        yield;
    }
}

// 自機狙い弾 - Phase 2/3で登場
task AimedPattern(boss) {
    let timer = 0;

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let bx = ObjEnemy_GetX(boss);
        let by = ObjEnemy_GetY(boss);
        let life = ObjEnemy_GetInfo(boss, INFO_LIFE);

        if(life <= 2000) {
            let n        = life > 1000 ? 3 : 5;
            let spread   = life > 1000 ? 22 : 18;
            let speed    = life > 1000 ? 2.8 : 3.5;
            let interval = life > 1000 ? 70 : 50;

            if(timer % interval == 0) {
                AimedNWay(bx, by, n, spread, speed, 1);
            }
        }

        timer++;
        yield;
    }
}

// 自機狙いN-Way散弾
function AimedNWay(x, y, n, spread, speed, graphic) {
    let baseAngle = atan2(GetPlayerY() - y, GetPlayerX() - x);
    loop(n) {
        let offset = (loop_index - (n - 1) / 2.0) * spread;
        CreateShotA(x, y, speed, baseAngle + offset, graphic, 4);
    }
}
