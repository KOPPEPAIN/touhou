#TouhouDanmakufu[Single]
#Title["桜符「散る花びら」"]
#Text["大型弾の花びらと降り注ぐ散弾の複合"]

// Phase 1 (HP 3000-2000): 16方向ゆっくり花びら弾 + 下方向散弾
// Phase 2 (HP 2000-1000): 花びら弾(高速) + 自機狙い3-Way
// Phase 3 (HP 1000-  0 ): 花びら弾(最速) + 自機狙い5-Way

task main {
    let boss = ObjEnemy_Create();
    ObjEnemy_SetPosition(boss, 192, 110);
    ObjEnemy_SetLife(boss, 3000);
    ObjEnemy_SetCollisionA(boss, 22);
    ObjEnemy_SetTimer(boss, 50);

    BossGlide(boss);
    PetalPattern(boss);
    ScatterPattern(boss);
    AimedPattern(boss);

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        yield;
    }
    Obj_Delete(boss);
}

task BossGlide(boss) {
    let t = 0;
    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        ObjEnemy_SetPosition(boss,
            192 + 90 * sin(t * 0.8),
            110 + 15 * sin(t * 1.5));
        t++;
        yield;
    }
}

task PetalPattern(boss) {
    let timer    = 0;
    let petalAngle = 0;

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let bx   = ObjEnemy_GetX(boss);
        let by   = ObjEnemy_GetY(boss);
        let life = ObjEnemy_GetInfo(boss, INFO_LIFE);

        let interval = 20;
        let speed    = 1.2;
        if(life <= 2000) { interval = 16; speed = 1.6; }
        if(life <= 1000) { interval = 12; speed = 2.0; }

        if(timer % interval == 0) {
            loop(8) {
                let a = petalAngle + loop_index * 45;
                CreateShotA(bx, by, speed, a,        2, 6);
                CreateShotA(bx, by, speed, a + 22.5, 2, 6);
            }
            petalAngle += 5;
        }

        timer++;
        yield;
    }
}

task ScatterPattern(boss) {
    let timer = 0;

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let bx   = ObjEnemy_GetX(boss);
        let by   = ObjEnemy_GetY(boss);
        let life = ObjEnemy_GetInfo(boss, INFO_LIFE);

        if(life > 2000 && timer % 8 == 0) {
            loop(5) {
                CreateShotA(bx, by, rand(1.5, 2.5), 90 + rand(-35, 35), 4, 0);
            }
        }

        timer++;
        yield;
    }
}

task AimedPattern(boss) {
    let timer = 0;

    while(ObjEnemy_GetInfo(boss, INFO_LIFE) > 0) {
        let bx   = ObjEnemy_GetX(boss);
        let by   = ObjEnemy_GetY(boss);
        let life = ObjEnemy_GetInfo(boss, INFO_LIFE);

        if(life <= 2000) {
            let n        = 3;
            let spread   = 22;
            let speed    = 2.8;
            let interval = 70;
            if(life <= 1000) { n = 5; spread = 18; speed = 3.5; interval = 50; }

            if(timer % interval == 0) {
                AimedNWay(bx, by, n, spread, speed, 1);
            }
        }

        timer++;
        yield;
    }
}

function AimedNWay(x, y, n, spread, speed, graphic) {
    let base = atan2(GetPlayerY() - y, GetPlayerX() - x);
    loop(n) {
        let offset = (loop_index - (n - 1) / 2.0) * spread;
        CreateShotA(x, y, speed, base + offset, graphic, 4);
    }
}
