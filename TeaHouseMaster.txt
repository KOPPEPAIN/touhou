// ============================================================
//  茶館の主 ～七彩喫茶夢幻譚～ (th_dnh_ph3 用)
//
//  このスクリプトは th_dnh_help_v3 の関数リファレンス
//  (func_common_std / func_common_obj_dx / func_common_obj_stg /
//   func_common_stg / syntax / other_spec) を直接参照して書き直して
//  あります。三項演算子(? :)・continue・const・TWait・
//  GetLoopIterator・Obj_GetX/Obj_SetAlpha・ObjEnemy_SetX/SetTimer
//  など、この処理系に存在しない構文/関数は使用していません。
//
//  弾グラフィックは th_dnh 既定弾 (DS_*) を使用します。
// ============================================================

let objEnemy = ObjEnemy_Create(OBJ_ENEMY);

let stageW = GetStgFrameWidth();
let stageH = GetStgFrameHeight();
let cx = stageW / 2;
let cy = stageH / 2;

let LIFE_P1 = 6000;
let LIFE_P2 = 6500;
let LIFE_P3 = 6500;
let LIFE_P4 = 7000;
let LIFE_P5 = 7000;
let LIFE_P6 = 7500;
let LIFE_P7 = 9000;

// フェーズ4/6の常駐サブタスク(DragonBand/MilkArm)を止めるフラグ
let bgTaskActive = true;

// 既定弾グラフィック割り当て (other_spec.html #DefaultShot 準拠)
let COL_LEAF_LIGHT = DS_BALL_M_GREEN;
let COL_LEAF_DEEP  = DS_BALL_L_GREEN;
let COL_GOLD       = DS_STAR_S_YELLOW;
let COL_MATCHA     = DS_BALL_M_GREEN;
let COL_AMBER      = DS_BALL_M_ORANGE;
let COL_MILKWHITE  = DS_BALL_M_WHITE;
let COL_BROWN      = DS_BALL_S_ORANGE;
let COL_TEAL       = DS_BALL_S_SKY;
let COL_GRAYBLUE   = DS_BALL_S_BLUE;
let COL_SPARK      = DS_STAR_S_ORANGE;
let COL_REDEMBER   = DS_BALL_M_RED;
let COL_PINK       = DS_BALL_S_PURPLE;

// ============================================================
// 汎用ヘルパー
// ============================================================
function Choose(let cond, let a, let b)
{
    if(cond) { return a; }
    return b;
}

function LifeRate(let total)
{
    return ObjEnemy_GetInfo(objEnemy, INFO_LIFE) / total * 100;
}

function BX() { return ObjMove_GetX(objEnemy); }
function BY() { return ObjMove_GetY(objEnemy); }
function PX() { return GetPlayerX(); }
function PY() { return GetPlayerY(); }

function Finish()
{
    DeleteShotAll(TYPE_ALL, TYPE_IMMEDIATE);
    ObjEnemy_SetLife(objEnemy, 0);
    Obj_Delete(objEnemy);
}

// ============================================================
// メイン制御
// ============================================================
@Initialize
{
    ObjMove_SetPosition(objEnemy, cx, 120);
    ObjEnemy_SetLife(objEnemy, LIFE_P1);
    ObjEnemy_SetDamageRate(objEnemy, 100, 100);
    ObjEnemy_SetIntersectionCircleToShot(objEnemy, 0, 0, 42);
    ObjEnemy_SetIntersectionCircleToPlayer(objEnemy, 0, 0, 42);
    ObjEnemy_Regist(objEnemy);

    MainSequence();
}

@MainLoop
{
    yield;
}

@Finalize
{
}

// ============================================================
// フェーズ進行 (全フェーズを1つのタスク内に直列で記述する。
// タスク呼び出しは同時並行実行になるため、フェーズを順番に
// 待つにはこのように1本のタスクの中に書く必要がある)
// ============================================================
task MainSequence
{
    loop(30) { yield; }

    // ===================================================
    // フェーズ1: 煎茶・若葉の渦
    // ===================================================
    let leafCount = 12;
    let baseAngIn = 0;
    let baseAngOut = 0;
    let dirIn = 1;
    let dirOut = 1;
    let flipTimer1 = 0;
    let flipping1 = false;
    let frameT1 = 0;

    while(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) > 0)
    {
        let rate = LifeRate(LIFE_P1);
        let col = Choose(rate <= 70, COL_LEAF_DEEP, COL_LEAF_LIGHT);
        let interval = Choose(rate <= 10, 20, 34);
        let spd = Choose(rate <= 10, 1.6, 2.0);

        if(rate <= 70)
        {
            flipTimer1++;
            if(flipTimer1 >= 180)
            {
                flipping1 = true;
                let shake = 0;
                while(shake < 18)
                {
                    ObjMove_SetX(objEnemy, cx + 6 * sin(shake * 60));
                    shake++;
                    yield;
                }
                dirIn *= -1;
                dirOut *= -1;
                flipTimer1 = 0;
                flipping1 = false;
            }
        }

        if(!flipping1 && modc(frameT1, interval) == 0)
        {
            let bx = BX();
            let by = BY();
            loop(leafCount)
            {
                baseAngIn += 360 / leafCount;
                let a = baseAngIn * dirIn;
                let obj1 = ObjShot_Create(OBJ_SHOT);
                ObjShot_SetGraphic(obj1, col);
                ObjMove_SetPosition(obj1, bx + 40 * cos(a), by + 40 * sin(a));
                ObjMove_SetSpeed(obj1, spd * 0.8);
                ObjMove_SetAngle(obj1, a);
                ObjShot_SetIntersectionCircleA1(obj1, 5);
                ObjShot_SetDelay(obj1, 8);
                ObjShot_SetAutoDelete(obj1, true);
                ObjShot_Regist(obj1);
            }
            loop(leafCount)
            {
                baseAngOut += 360 / leafCount;
                let a = -baseAngOut * dirOut;
                let obj2 = ObjShot_Create(OBJ_SHOT);
                ObjShot_SetGraphic(obj2, col);
                ObjMove_SetPosition(obj2, bx + 70 * cos(a), by + 70 * sin(a));
                ObjMove_SetSpeed(obj2, spd * 1.25);
                ObjMove_SetAngle(obj2, a);
                ObjShot_SetIntersectionCircleA1(obj2, 5);
                ObjShot_SetDelay(obj2, 8);
                ObjShot_SetAutoDelete(obj2, true);
                ObjShot_Regist(obj2);
            }
        }

        if(rate <= 40 && modc(frameT1, 50) == 0)
        {
            let obj3 = ObjShot_Create(OBJ_SHOT);
            ObjShot_SetGraphic(obj3, COL_GOLD);
            ObjMove_SetPosition(obj3, BX(), BY());
            ObjMove_SetSpeed(obj3, 3.5);
            ObjMove_SetAngle(obj3, 90);
            ObjShot_SetIntersectionCircleA1(obj3, 3);
            ObjShot_SetDelay(obj3, 6);
            ObjShot_SetAutoDelete(obj3, true);
            ObjShot_Regist(obj3);
        }

        frameT1++;
        yield;
    }
    if(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) <= 0) { Finish(); return; }

    // ===================================================
    // フェーズ2: 抹茶・泡沫翠環
    // ===================================================
    ObjEnemy_SetLife(objEnemy, LIFE_P2);
    let ringRot2 = 0;
    let frameT2 = 0;

    while(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) > 0)
    {
        let rate = LifeRate(LIFE_P2);

        ObjMove_SetX(objEnemy, cx + 90 * sin(frameT2 * 0.7));

        if(rate <= 40) { ringRot2 += 0.4; }

        if(modc(frameT2, 130) == 0)
        {
            let gapAngle = ringRot2 + 45;
            let outerN = 16;
            let innerN = 20;
            let bx = BX();
            let by = BY();

            ascent(let idx in 0..(outerN - 1)) loop
            {
                let a = ringRot2 + idx * (360 / outerN);
                let isGap = absolute(modc(a - gapAngle + 720, 360)) < (360 / outerN);
                if(!isGap) { CreateBubble(bx, by, 120, a, 0.6, true, rate); }
            }
            ascent(let idx in 0..(innerN - 1)) loop
            {
                let a = -ringRot2 + idx * (360 / innerN);
                let isGap = absolute(modc(a - gapAngle + 720, 360)) < (360 / innerN);
                if(!isGap) { CreateBubble(bx, by, 60, a, 1.1, false, rate); }
            }
        }

        if(rate <= 10 && modc(frameT2, 8) == 0)
        {
            let a = frameT2 * 6;
            CreateBubble(BX(), BY(), 55, a, 1.2, false, rate);
        }
        if(rate <= 10 && modc(frameT2, 40) == 0)
        {
            let a = rand(0, 360);
            CreateBubble(BX(), BY(), 130, a, 0.55, true, rate);
        }

        frameT2++;
        yield;
    }
    if(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) <= 0) { Finish(); return; }

    // ===================================================
    // フェーズ3: 紅茶・琥珀の雫雨
    // ===================================================
    ObjEnemy_SetLife(objEnemy, LIFE_P3);
    let waveDir3 = 1;
    let waveX3 = 0;
    let diagFlip3 = 1;
    let frameT3 = 0;

    while(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) > 0)
    {
        let rate = LifeRate(LIFE_P3);

        if(modc(frameT3, 12) == 0)
        {
            waveX3 += waveDir3 * 24;
            if(waveX3 > stageW - 20 || waveX3 < 20) { waveDir3 *= -1; }

            let big = modc(frameT3, 96) == 0;
            let dropAng = 90;

            if(rate <= 70 && modc(frameT3, 8) == 0)
            {
                diagFlip3 = Choose(modc(int(frameT3 / 240), 2) == 0, 1, -1);
                let sx = Choose(diagFlip3 == 1, rand(0, stageW * 0.3), rand(stageW * 0.7, stageW));
                CreateDrop(sx, -10, dropAng + diagFlip3 * 20, big, rate);
            }
            else
            {
                CreateDrop(waveX3, -10, dropAng, big, rate);
            }
        }

        frameT3++;
        yield;
    }
    if(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) <= 0) { Finish(); return; }

    // ===================================================
    // フェーズ4: 烏龍茶・龍の茶煙
    // ===================================================
    ObjEnemy_SetLife(objEnemy, LIFE_P4);
    bgTaskActive = true;
    DragonBand(1);
    DragonBand(-1);
    let frameT4 = 0;

    while(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) > 0)
    {
        let rate = LifeRate(LIFE_P4);
        ObjMove_SetX(objEnemy, cx + 60 * sin(frameT4 * 0.5));

        if(rate <= 40 && modc(frameT4, 130) == 0)
        {
            let obj = ObjShot_Create(OBJ_SHOT);
            ObjShot_SetGraphic(obj, COL_GRAYBLUE);
            ObjMove_SetPosition(obj, PX(), 0);
            ObjMove_SetSpeed(obj, 1.4);
            ObjMove_SetAngle(obj, 90);
            ObjShot_SetIntersectionCircleA1(obj, 14);
            ObjShot_SetDelay(obj, 10);
            ObjShot_SetAutoDelete(obj, true);
            ObjShot_Regist(obj);
            DragonHeadTail(obj);
        }
        frameT4++;
        yield;
    }
    bgTaskActive = false;
    if(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) <= 0) { Finish(); return; }

    // ===================================================
    // フェーズ5: ほうじ茶・焙煎火花
    // ===================================================
    ObjEnemy_SetLife(objEnemy, LIFE_P5);
    let quadrant5 = 0;
    let frameT5 = 0;

    while(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) > 0)
    {
        let rate = LifeRate(LIFE_P5);

        if(rate > 10 && modc(frameT5, 70) == 0)
        {
            let startA = quadrant5 * 90;
            ascent(let idx in 0..7) loop
            {
                let a = startA + idx * (90 / 8);
                CreateEmber(BX(), BY(), a, rate, 90);
            }
            quadrant5 = modc(quadrant5 + 1, 4);
        }

        if(rate <= 40 && modc(frameT5, 45) == 0)
        {
            let hx = rand(40, stageW - 40);
            let obj = ObjShot_Create(OBJ_SHOT);
            ObjShot_SetGraphic(obj, COL_REDEMBER);
            ObjMove_SetPosition(obj, hx, stageH + 10);
            ObjMove_SetSpeed(obj, 1.2);
            ObjMove_SetAngle(obj, 270);
            ObjShot_SetIntersectionCircleA1(obj, 4);
            ObjShot_SetAutoDelete(obj, true);
            ObjShot_Regist(obj);
            RisingHeat(obj);
        }

        if(rate <= 10 && modc(frameT5, 12) == 0)
        {
            let flowAng5 = 90 * modc(int(frameT5 / 90), 4);
            ascent(let idx in 0..15) loop
            {
                let a = idx * (360 / 16) + frameT5 * 2;
                CreateEmber(BX(), BY(), a, rate, flowAng5);
            }
        }

        frameT5++;
        yield;
    }
    if(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) <= 0) { Finish(); return; }

    // ===================================================
    // フェーズ6: ミルクティー・白霞の螺旋
    // ===================================================
    ObjEnemy_SetLife(objEnemy, LIFE_P6);
    bgTaskActive = true;
    MilkArm(0, 1);
    MilkArm(90, 1);
    MilkArm(180, -1);
    MilkArm(270, -1);
    let frameT6 = 0;

    while(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) > 0)
    {
        let rate = LifeRate(LIFE_P6);

        if(rate <= 40 && modc(frameT6, 55) == 0)
        {
            ascent(let idx in 0..5) loop
            {
                let a = idx * 60 + frameT6;
                let obj = ObjShot_Create(OBJ_SHOT);
                ObjShot_SetGraphic(obj, COL_PINK);
                ObjMove_SetPosition(obj, BX(), BY());
                ObjMove_SetSpeed(obj, 1.4);
                ObjMove_SetAngle(obj, a);
                ObjShot_SetIntersectionCircleA1(obj, 3);
                ObjShot_SetAutoDelete(obj, true);
                ObjShot_Regist(obj);
                let spin = Choose(modc(idx, 2) == 0, 1, -1);
                PetalSpin(obj, spin);
            }
        }

        frameT6++;
        yield;
    }
    bgTaskActive = false;
    if(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) <= 0) { Finish(); return; }

    // ===================================================
    // フェーズ7: 茶会・七彩千客万来 (最終)
    // ===================================================
    ObjEnemy_SetLife(objEnemy, LIFE_P7);
    ObjMove_SetPosition(objEnemy, cx, cy - 60);
    let cycleTimer7 = 0;
    let ringRotation7 = 0;
    let frameT7 = 0;

    while(ObjEnemy_GetInfo(objEnemy, INFO_LIFE) > 0)
    {
        let rate = LifeRate(LIFE_P7);

        if(rate > 10)
        {
            let cycleLen = Choose(rate <= 20, 360, 480);
            let quarter = cycleLen / 4;
            cycleTimer7 = modc(cycleTimer7 + 1, cycleLen);

            if(rate <= 40) { ringRotation7 += 0.15; }

            let seg = int(cycleTimer7 / quarter);
            let bx = BX() + 40 * cos(ringRotation7);
            let by = BY() + 40 * sin(ringRotation7);

            if(seg == 0 && modc(cycleTimer7, 6) == 0)
            {
                ascent(let idx in 0..1) loop
                {
                    let a = ringRotation7 * 57.3 + idx * 180 + cycleTimer7 * 3;
                    let obj = ObjShot_Create(OBJ_SHOT);
                    ObjShot_SetGraphic(obj, COL_LEAF_LIGHT);
                    ObjMove_SetPosition(obj, bx, by);
                    ObjMove_SetSpeed(obj, 1.8);
                    ObjMove_SetAngle(obj, a);
                    ObjShot_SetIntersectionCircleA1(obj, 3);
                    ObjShot_SetAutoDelete(obj, true);
                    ObjShot_Regist(obj);
                }
            }
            else if(seg == 1 && modc(cycleTimer7, 14) == 0)
            {
                let sx = rand(0, stageW);
                CreateDrop(sx, -10, 90, false, 100);
            }
            else if(seg == 2 && modc(cycleTimer7, 5) == 0)
            {
                let a = ringRotation7 * 57.3 + cycleTimer7 * 4;
                let obj = ObjShot_Create(OBJ_SHOT);
                ObjShot_SetGraphic(obj, COL_MILKWHITE);
                ObjMove_SetPosition(obj, bx, by);
                ObjMove_SetSpeed(obj, 1.5);
                ObjMove_SetAngle(obj, a);
                ObjShot_SetIntersectionCircleA1(obj, 3);
                ObjShot_SetAutoDelete(obj, true);
                ObjShot_Regist(obj);
            }
            else if(seg == 3 && modc(cycleTimer7, 8) == 0)
            {
                CreateEmber(BX(), BY(), rand(0, 360), 50, 90);
                CreateSmokeBullet(BX() + 60 * cos(cycleTimer7), BY() + 60 * sin(cycleTimer7), 1, cycleTimer7);
            }

            if(rate <= 70 && cycleTimer7 == cycleLen - 1)
            {
                let gap = rand(0, 360);
                ascent(let idx in 0..11) loop
                {
                    let a = idx * 30;
                    let isGap1 = absolute(modc(a - gap + 720, 360)) < 30;
                    let isGap2 = absolute(modc(a - gap + 540 + 720, 360)) < 30;
                    if(!isGap1 && !isGap2) { CreateBubble(BX(), BY(), 100, a, 0.7, true, 100); }
                }
            }
        }
        else
        {
            ObjMove_SetPosition(objEnemy, cx, cy - 40);
            FinalDesperation(frameT7);
        }

        frameT7++;
        yield;
    }

    Finish();
}

// ============================================================
// フェーズ4: 烏龍茶 常駐サブタスク
// ============================================================
task DragonBand(let side)
{
    let phase = Choose(side > 0, 0, 180);
    let waveAmp = 140;
    let t = 0;
    while(bgTaskActive && ObjEnemy_GetInfo(objEnemy, INFO_LIFE) > 0)
    {
        let rate = LifeRate(LIFE_P4);
        let branch = rate <= 70 && modc(t, 400) < 60;

        if(modc(t, 5) == 0)
        {
            let bandX = cx + side * 100 + waveAmp * sin(t * 0.02 + phase);
            let bandY = modc(t * 1.5, stageH + 100) - 50;

            CreateSmokeBullet(bandX, bandY, side, t);
            if(branch) { CreateSmokeBullet(bandX + side * 40, bandY - 20, side, t); }

            if(modc(t, 20) == 0)
            {
                let d = ObjShot_Create(OBJ_SHOT);
                ObjShot_SetGraphic(d, COL_GOLD);
                ObjMove_SetPosition(d, bandX, bandY);
                ObjMove_SetSpeed(d, 3.2);
                ObjMove_SetAngle(d, 90);
                ObjShot_SetIntersectionCircleA1(d, 2);
                ObjShot_SetAutoDelete(d, true);
                ObjShot_Regist(d);
            }
        }

        if(rate <= 10)
        {
            waveAmp = stageW * 0.4;
            phase += side * 1.2;
        }

        t++;
        yield;
    }
}

function CreateSmokeBullet(let x, let y, let side, let t)
{
    let obj = ObjShot_Create(OBJ_SHOT);
    ObjShot_SetGraphic(obj, Choose(modc(t, 2) == 0, COL_TEAL, COL_GRAYBLUE));
    ObjMove_SetPosition(obj, x, y);
    ObjMove_SetSpeed(obj, 1.3);
    ObjMove_SetAngle(obj, 90 + side * 15);
    ObjShot_SetIntersectionCircleA1(obj, 4);
    ObjShot_SetDelay(obj, 6);
    ObjShot_SetAutoDelete(obj, true);
    ObjShot_Regist(obj);
}

task DragonHeadTail(let head)
{
    let t = 0;
    while(t < 300 && ObjMove_GetY(head) < stageH + 20)
    {
        if(modc(t, 6) == 0)
        {
            let tail = ObjShot_Create(OBJ_SHOT);
            ObjShot_SetGraphic(tail, COL_TEAL);
            ObjMove_SetPosition(tail, ObjMove_GetX(head), ObjMove_GetY(head));
            ObjMove_SetSpeed(tail, 1.0);
            ObjMove_SetAngle(tail, 90 + 20 * sin(t * 10));
            ObjShot_SetIntersectionCircleA1(tail, 3);
            ObjShot_SetAutoDelete(tail, true);
            ObjShot_Regist(tail);
        }
        t++;
        yield;
    }
    Obj_Delete(head);
}

// フェーズ2: 抹茶 弾生成タスク
// ============================================================
task CreateBubble(let bx, let by, let dist, let ang, let spd, let isBig, let rate)
{
    let obj = ObjShot_Create(OBJ_SHOT);
    let size = Choose(isBig, 10, 5);
    ObjShot_SetGraphic(obj, COL_MATCHA);
    ObjMove_SetPosition(obj, bx + dist * cos(ang), by + dist * sin(ang));
    ObjMove_SetSpeed(obj, 0);
    ObjMove_SetAngle(obj, ang);
    ObjShot_SetIntersectionCircleA1(obj, size * 0.5 + 2);
    ObjShot_SetAutoDelete(obj, true);
    ObjShot_Regist(obj);

    let blink = 0;
    while(blink < 24)
    {
        ObjRender_SetAlpha(obj, 128 + 100 * sin(blink * 40));
        blink++;
        yield;
    }
    ObjRender_SetAlpha(obj, 255);

    let tx = PX();
    let ty = PY();
    let dx = tx - ObjMove_GetX(obj);
    let dy = ty - ObjMove_GetY(obj);
    let baseAng = atan2(dy, dx);
    let curve = Choose(isBig, -0.3, 0.5);
    let t = 0;
    let split = false;
    let splitAt = 75;

    while(t < 400)
    {
        let curAng = baseAng + curve * sin(t * 0.03) * 20;
        ObjMove_SetAngle(obj, curAng);
        ObjMove_SetSpeed(obj, spd);

        if(rate <= 70 && !split && t == splitAt - 15)
        {
            let f = 0;
            while(f < 15)
            {
                ObjRender_SetAlpha(obj, 200 + 55 * sin(f * 80));
                f++;
                yield;
            }
        }
        if(rate <= 70 && !split && t == splitAt)
        {
            split = true;
            let curAng2 = ObjMove_GetAngle(obj);
            SpawnSmallBubble(ObjMove_GetX(obj), ObjMove_GetY(obj), curAng2 - 18, spd);
            SpawnSmallBubble(ObjMove_GetX(obj), ObjMove_GetY(obj), curAng2 + 18, spd);
        }

        let bx2 = ObjMove_GetX(obj);
        let by2 = ObjMove_GetY(obj);
        if(bx2 < -32 || bx2 > stageW + 32 || by2 < -32 || by2 > stageH + 32) { break; }
        t++;
        yield;
    }
    Obj_Delete(obj);
}

function SpawnSmallBubble(let x, let y, let ang, let spd)
{
    let obj = ObjShot_Create(OBJ_SHOT);
    ObjShot_SetGraphic(obj, COL_MATCHA);
    ObjMove_SetPosition(obj, x, y);
    ObjMove_SetSpeed(obj, spd * 1.1);
    ObjMove_SetAngle(obj, ang);
    ObjShot_SetIntersectionCircleA1(obj, 3);
    ObjShot_SetDelay(obj, 4);
    ObjShot_SetAutoDelete(obj, true);
    ObjShot_Regist(obj);
}

// ============================================================
// フェーズ3: 紅茶 弾生成タスク
// ============================================================
task CreateDrop(let sx, let sy, let ang, let isBig, let rate)
{
    let guide = ObjShot_Create(OBJ_SHOT);
    ObjShot_SetGraphic(guide, COL_AMBER);
    ObjMove_SetPosition(guide, sx, stageH - 10);
    ObjRender_SetAlpha(guide, 90);
    ObjShot_SetIntersectionCircleA1(guide, 1);
    ObjMove_SetSpeed(guide, 0);
    ObjMove_SetAngle(guide, 0);
    ObjShot_SetAutoDelete(guide, true);
    ObjShot_Regist(guide);
    loop(20) { yield; }
    Obj_Delete(guide);

    let obj = ObjShot_Create(OBJ_SHOT);
    ObjShot_SetGraphic(obj, COL_AMBER);
    ObjMove_SetPosition(obj, sx, sy);
    let spd = Choose(isBig, 1.3, 1.6);
    ObjMove_SetSpeed(obj, spd);
    ObjMove_SetAngle(obj, ang);
    ObjShot_SetIntersectionCircleA1(obj, Choose(isBig, 8, 4));
    ObjShot_SetDelay(obj, 6);
    ObjShot_SetAutoDelete(obj, true);
    ObjShot_Regist(obj);

    let accelerated = false;
    let t = 0;
    while(t < 500)
    {
        let y = ObjMove_GetY(obj);
        if(!accelerated && y > stageH * 0.45)
        {
            accelerated = true;
            ObjMove_SetSpeed(obj, spd * 2.4);
            if(isBig)
            {
                ascent(let idx in 0..4) loop
                {
                    let pAng = 90 + (idx - 2) * 22;
                    let p = ObjShot_Create(OBJ_SHOT);
                    ObjShot_SetGraphic(p, COL_BROWN);
                    ObjMove_SetPosition(p, ObjMove_GetX(obj), ObjMove_GetY(obj));
                    ObjMove_SetSpeed(p, 1.8);
                    ObjMove_SetAngle(p, pAng);
                    ObjShot_SetIntersectionCircleA1(p, 2);
                    ObjShot_SetAutoDelete(p, true);
                    ObjShot_Regist(p);
                    DecayParticle(p);
                }
            }
        }

        if(rate <= 40 && y >= stageH - 16)
        {
            let dir = Choose(ObjMove_GetX(obj) < cx, 1, -1);
            let r1 = ObjShot_Create(OBJ_SHOT);
            ObjShot_SetGraphic(r1, COL_MILKWHITE);
            ObjMove_SetPosition(r1, ObjMove_GetX(obj), stageH - 12);
            ObjMove_SetSpeed(r1, 0.7);
            ObjMove_SetAngle(r1, Choose(dir > 0, 0, 180));
            ObjShot_SetIntersectionCircleA1(r1, 3);
            ObjShot_SetAutoDelete(r1, true);
            ObjShot_Regist(r1);
            break;
        }

        if(y > stageH + 20 || ObjMove_GetX(obj) < -20 || ObjMove_GetX(obj) > stageW + 20) { break; }
        t++;
        yield;
    }
    Obj_Delete(obj);
}

task DecayParticle(let obj2)
{
    let life = 0;
    while(life < 40)
    {
        ObjMove_SetSpeed(obj2, ObjMove_GetSpeed(obj2) * 0.94);
        life++;
        yield;
    }
    Obj_Delete(obj2);
}

task CreateEmber(let bx, let by, let ang, let rate, let flowAng)
{
    let obj = ObjShot_Create(OBJ_SHOT);
    ObjShot_SetGraphic(obj, COL_SPARK);
    ObjMove_SetPosition(obj, bx + 40 * cos(ang), by + 40 * sin(ang));
    ObjMove_SetSpeed(obj, 0);
    ObjMove_SetAngle(obj, ang);
    ObjShot_SetIntersectionCircleA1(obj, 3);
    ObjShot_SetAutoDelete(obj, true);
    ObjShot_Regist(obj);

    loop(20) { yield; }
    ObjMove_SetSpeed(obj, Choose(rate <= 10, 3.0, 2.2));
    ObjMove_SetAngle(obj, ang);
    loop(14) { yield; }

    ObjMove_SetSpeed(obj, 0.3);
    if(rate <= 70)
    {
        ascent(let idx in 0..3) loop
        {
            let sparkAng = idx * 90;
            let sp = ObjShot_Create(OBJ_SHOT);
            ObjShot_SetGraphic(sp, COL_SPARK);
            ObjMove_SetPosition(sp, ObjMove_GetX(obj), ObjMove_GetY(obj));
            ObjMove_SetSpeed(sp, 1.0);
            ObjMove_SetAngle(sp, sparkAng);
            ObjShot_SetIntersectionCircleA1(sp, 1.5);
            ObjShot_SetAutoDelete(sp, true);
            ObjShot_Regist(sp);
        }
    }

    let t = 0;
    while(t < 200)
    {
        ObjMove_SetAngle(obj, 90 + (flowAng - 90) * 0.3);
        ObjMove_SetSpeed(obj, 0.9);
        if(ObjMove_GetY(obj) > stageH + 10) { break; }
        t++;
        yield;
    }
    Obj_Delete(obj);
}

task RisingHeat(let o)
{
    let s = 0;
    while(ObjMove_GetY(o) > -20)
    {
        ObjMove_SetX(o, ObjMove_GetX(o) + 1.6 * sin(s * 10));
        s++;
        yield;
    }
    Obj_Delete(o);
}

// フェーズ6: ミルクティー 常駐サブタスク
// ============================================================
task MilkArm(let startAng, let dir)
{
    let ang = startAng;
    let radius = 30;
    let t = 0;
    while(bgTaskActive && ObjEnemy_GetInfo(objEnemy, INFO_LIFE) > 0)
    {
        let rate = LifeRate(LIFE_P6);

        if(rate <= 10)
        {
            let cyc = modc(t, 90);
            radius = Choose(cyc < 30, 30 - cyc, (cyc - 30) * 3);
        }
        else
        {
            radius += 1.6;
            if(radius > stageW * 0.55) { radius = 30; }
        }

        ang += dir * 1.5;

        if(modc(t, 4) == 0)
        {
            let bx = BX() + radius * cos(ang);
            let by = BY() + radius * sin(ang);
            let obj = ObjShot_Create(OBJ_SHOT);
            ObjShot_SetGraphic(obj, COL_MILKWHITE);
            ObjMove_SetPosition(obj, bx, by);
            ObjMove_SetSpeed(obj, 1.0);
            ObjMove_SetAngle(obj, ang);
            ObjShot_SetIntersectionCircleA1(obj, 4);
            ObjShot_SetDelay(obj, 6);
            ObjShot_SetAutoDelete(obj, true);
            ObjShot_Regist(obj);

            let obj2 = ObjShot_Create(OBJ_SHOT);
            ObjShot_SetGraphic(obj2, COL_BROWN);
            ObjMove_SetPosition(obj2, BX() + (radius * 0.6) * cos(ang + 15), BY() + (radius * 0.6) * sin(ang + 15));
            ObjMove_SetSpeed(obj2, 0.9);
            ObjMove_SetAngle(obj2, ang + 15);
            ObjShot_SetIntersectionCircleA1(obj2, 2.5);
            ObjShot_SetAutoDelete(obj2, true);
            ObjShot_Regist(obj2);

            if(rate <= 70) { MilkDropColorChange(obj2, dir); }
        }

        t++;
        yield;
    }
}

task MilkDropColorChange(let o, let dir)
{
    loop(90) { yield; }
    let f = 0;
    while(f < 10)
    {
        ObjRender_SetAlpha(o, 200 + 55 * sin(f * 90));
        f++;
        yield;
    }
    ObjShot_SetGraphic(o, COL_MILKWHITE);
    ObjMove_SetAngle(o, ObjMove_GetAngle(o) - dir * 25);
}

task PetalSpin(let o, let spin)
{
    let s = 0;
    while(s < 240)
    {
        ObjMove_SetAngle(o, ObjMove_GetAngle(o) - spin * 0.6);
        let x = ObjMove_GetX(o);
        let y = ObjMove_GetY(o);
        if(x < -20 || x > stageW + 20 || y < -20 || y > stageH + 20) { break; }
        s++;
        yield;
    }
    Obj_Delete(o);
}

// フェーズ7: 茶会 最終発狂
// ============================================================
function FinalDesperation(let frameT)
{
    let gapBase = frameT * 0.6;
    let layers = 7;
    let colArr = [COL_LEAF_LIGHT, DS_BALL_M_YELLOW, COL_AMBER, COL_TEAL, COL_REDEMBER, COL_MILKWHITE, DS_BALL_M_PURPLE];
    let spd = 0.85;

    if(modc(frameT, 10) == 0)
    {
        ascent(let li in 0..6) loop
        {
            let layerRadius = 220 - li * 26;
            let gapAng = gapBase + li * (360 / layers);
            let n = 10 + li;
            ascent(let idx in 0..(n - 1)) loop
            {
                let a = idx * (360 / n);
                let isGap = absolute(modc(a - gapAng + 720, 360)) < (360 / n);
                if(!isGap)
                {
                    let obj = ObjShot_Create(OBJ_SHOT);
                    ObjShot_SetGraphic(obj, colArr[li]);
                    ObjMove_SetPosition(obj, BX() + layerRadius * cos(a), BY() + layerRadius * sin(a));
                    ObjMove_SetSpeed(obj, spd);
                    ObjMove_SetAngle(obj, a + 180);
                    ObjShot_SetIntersectionCircleA1(obj, 3);
                    ObjShot_SetDelay(obj, 8);
                    ObjShot_SetAutoDelete(obj, true);
                    ObjShot_Regist(obj);
                }
            }
        }
    }
}
