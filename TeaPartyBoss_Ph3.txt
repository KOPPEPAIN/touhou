//==================================================================
// 共通ユーティリティ
//==================================================================
function Wait(f) {
    loop(f) { yield; }
}

function AngleDiff(a, b) {
    let d = a - b;
    while (d > 180) { d = d - 360; }
    while (d < -180) { d = d + 360; }
    return d;
}

function AngleTo(x1, y1, x2, y2) {
    return atan2(y2 - y1, x2 - x1);
}

//==================================================================
// 弾グラフィックID
// ph3規定の弾セットは「形状(十の位)+色(一の位)」で構成されているため、
// 色8(白/灰)の弾をベースに ObjRender_SetColor で任意の色に着色して使う。
// 自分のリソースに合わせてIDだけ書き換えれば流用できます。
//==================================================================
let ID_BALL   = 8;   // 小玉
let ID_BIG    = 18;  // 大玉
let ID_STAR   = 28;  // 星
let ID_RICE   = 38;  // 米粒 → 茶葉に流用
let ID_NEEDLE = 48;  // 針 → 金の直線弾に流用
let ID_TEAR   = 68;  // 涙型 → 雫に流用
let ID_RING   = 78;  // 輪 → 泡に流用

//==================================================================
// グローバル状態
//==================================================================
let enemy = -1;
let baseX = 192;
let baseY = 120;
let hpPercent = 100;
let phaseMaxLife = 1;
let phaseActive = false;

// フェーズ1
let spiralDir = 1;
let spiralPaused = false;
let leafRingArr = [];

// フェーズ2
let ringGapAngle = 0;
let ringPhaseAngle = 0;

// フェーズ6
let spiralPulse = 1.0;

// フェーズ7
let haloAngleOffset = 0;

//==================================================================
// メイン
//==================================================================
task main {
    enemy = ObjEnemy_Create();
    ObjEnemy_Regist(enemy);
    ObjEnemy_SetX(enemy, baseX);
    ObjEnemy_SetY(enemy, baseY);
    ObjEnemy_SetCollisionA(enemy, 32);
    ObjEnemy_SetTimer(enemy, 0);
    // お好みで自前のグラフィックに変更可能（未設定でも動作します）:
    // ObjEnemy_SetGraphic(enemy, "./img/tea_boss.png");

    Phase1_SenchaWakaba();
    Phase2_MatchaHouwaSuikan();
    Phase3_KoukyaKohakuShizuku();
    Phase4_OolongRyuuChaen();
    Phase5_HoujichaBaisenHibana();
    Phase6_MilkTeaHakkaRasen();
    Phase7_ChakaiShichisaiSenkyakuBanrai();

    ObjEnemy_Delete(enemy);
}

//==================================================================
// フェーズ1: 煎茶・若葉の渦
//==================================================================
function Phase1_SenchaWakaba() {
    phaseMaxLife = 9000;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    ObjEnemy_SetTimer(enemy, 0);
    phaseActive = true;
    spiralDir = 1;
    spiralPaused = false;
    hpPercent = 100;

    leafRingArr = [];
    ascent(i in 0..12) {
        let sh = CreateShotA1(baseX, baseY, 0, i * 30, ID_RICE, 1);
        ObjRender_SetColor(sh, 170, 230, 90);
        leafRingArr = leafRingArr ~ [sh];
    }

    Phase1_LeafRingTask();
    Phase1_SpiralTask();
    Phase1_FlipTask();
    Phase1_GoldLineTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }

    phaseActive = false;
    Wait(6);
    ascent(i in 0..length(leafRingArr)) {
        if (!Obj_IsDeleted(leafRingArr[i])) { Obj_Delete(leafRingArr[i]); }
    }
}

task Phase1_LeafRingTask() {
    let ang = 0;
    while (phaseActive) {
        let bx = ObjEnemy_GetX(enemy);
        let by = ObjEnemy_GetY(enemy);
        let dark = hpPercent <= 70;
        ascent(i in 0..length(leafRingArr)) {
            if (!Obj_IsDeleted(leafRingArr[i])) {
                let a = ang + i * 30;
                ObjMove_SetPosition(leafRingArr[i], bx + 44 * cos(a), by + 44 * sin(a));
                if (dark) { ObjRender_SetColor(leafRingArr[i], 30, 90, 30); }
                else { ObjRender_SetColor(leafRingArr[i], 170, 230, 90); }
            }
        }
        ang = ang + spiralDir * 1.2;
        yield;
    }
}

task Phase1_SpiralTask() {
    let angIn = 0;
    let angOut = 0;
    while (phaseActive) {
        if (!spiralPaused) {
            let bx = ObjEnemy_GetX(enemy);
            let by = ObjEnemy_GetY(enemy);
            let dark = hpPercent <= 70;
            let r = 170; let g = 230; let b = 90;
            if (dark) { r = 30; g = 100; b = 30; }

            let sIn = CreateShotA1(bx, by, 1.6, angIn, ID_RICE, 1);
            ObjRender_SetColor(sIn, r, g, b);
            let sOut = CreateShotA1(bx, by, 2.3, angOut, ID_RICE, 1);
            ObjRender_SetColor(sOut, r, g, b);

            angIn = angIn + spiralDir * 24;
            angOut = angOut + spiralDir * 33;
        }
        Wait(hpPercent <= 10 ? 8 : 14);
    }
}

task Phase1_FlipTask() {
    while (phaseActive) {
        Wait(180);
        if (phaseActive && hpPercent <= 70) {
            spiralPaused = true;
            ascent(i in 0..12) {
                ObjEnemy_SetX(enemy, baseX + 5 * sin(i * 60));
                Wait(2);
            }
            ObjEnemy_SetX(enemy, baseX);
            spiralDir = -spiralDir;
            spiralPaused = false;
        }
    }
}

task Phase1_GoldLineTask() {
    while (phaseActive) {
        if (hpPercent <= 40) {
            let sh = CreateShotA1(ObjEnemy_GetX(enemy), ObjEnemy_GetY(enemy), 3.2, 90, ID_NEEDLE, 20);
            ObjRender_SetColor(sh, 255, 210, 90);
        }
        Wait(hpPercent <= 10 ? 26 : 46);
    }
}

// フェーズ2: 抹茶・泡沫翠環
//==================================================================
function Phase2_MatchaHouwaSuikan() {
    phaseMaxLife = 9500;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    ObjEnemy_SetTimer(enemy, 0);
    phaseActive = true;
    hpPercent = 100;
    ringGapAngle = 0;
    ringPhaseAngle = 0;

    Phase2_SwayTask();
    Phase2_RingTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
    ObjEnemy_SetX(enemy, baseX);
}

task Phase2_SwayTask() {
    let t = 0;
    while (phaseActive) {
        ObjEnemy_SetX(enemy, baseX + 36 * sin(t));
        t = t + 1.0;
        yield;
    }
}

task Phase2_RingTask() {
    while (phaseActive) {
        let bx = ObjEnemy_GetX(enemy);
        let by = ObjEnemy_GetY(enemy);
        ascent(i in 0..10) {
            let a = ringPhaseAngle + i * 36;
            if (abs(AngleDiff(a, ringGapAngle)) > 18) {
                Phase2_SpawnBubble(bx, by, a, 210, true);
            }
        }
        ascent(i in 0..14) {
            let a = ringPhaseAngle + i * (360.0 / 14.0) + 12;
            if (abs(AngleDiff(a, ringGapAngle)) > 13) {
                Phase2_SpawnBubble(bx, by, a, 130, false);
            }
        }
        if (hpPercent <= 40) {
            ringGapAngle = ringGapAngle + 1.4;
            ringPhaseAngle = ringPhaseAngle + 0.7;
        }
        Wait(hpPercent <= 10 ? 65 : 95);
    }
}

task Phase2_SpawnBubble(bx, by, angle, radius, isBig) {
    let x = bx + radius * cos(angle);
    let y = by + radius * sin(angle);
    let sh = CreateShotA1(x, y, 0, angle, ID_RING, 1);
    ObjRender_SetAlpha(sh, 90);
    ObjRender_SetColor(sh, 170, 230, 170);

    ascent(i in 0..16) {
        if (Obj_IsDeleted(sh)) { return; }
        ObjRender_SetAlpha(sh, 90 + i * 10);
        ObjRender_SetColor(sh, 60 + i * 7, 200, 60 + i * 7);
        Wait(2);
    }
    Wait(isBig ? 30 : 14);
    if (Obj_IsDeleted(sh)) { return; }

    let tx = GetPlayerX();
    let ty = GetPlayerY();
    let ang = AngleTo(ObjMove_GetX(sh), ObjMove_GetY(sh), tx, ty);
    ObjMove_SetAngle(sh, ang);
    ObjMove_SetSpeed(sh, isBig ? 1.1 : 1.9);
    let curveSign = 1;
    if (ObjMove_GetX(sh) < bx) { curveSign = -1; }
    ObjMove_SetAngularVelocity(sh, curveSign * 0.28);

    if (isBig && hpPercent <= 70) {
        Wait(50);
        if (!Obj_IsDeleted(sh)) {
            ascent(i in 0..8) {
                if (Obj_IsDeleted(sh)) { return; }
                ObjRender_SetColor(sh, 255, 255, 255);
                Wait(2);
                if (Obj_IsDeleted(sh)) { return; }
                ObjRender_SetColor(sh, 60, 200, 60);
                Wait(2);
            }
            if (!Obj_IsDeleted(sh)) {
                let sx = ObjMove_GetX(sh);
                let sy = ObjMove_GetY(sh);
                let bAng = ObjMove_GetAngle(sh);
                let bSpd = ObjMove_GetSpeed(sh);
                Obj_Delete(sh);
                let c1 = CreateShotA1(sx, sy, bSpd, bAng - 18, ID_RING, 1);
                ObjRender_SetColor(c1, 100, 220, 100);
                let c2 = CreateShotA1(sx, sy, bSpd, bAng + 18, ID_RING, 1);
                ObjRender_SetColor(c2, 100, 220, 100);
            }
        }
    }
}

// フェーズ3: 紅茶・琥珀の雫雨
//==================================================================
function Phase3_KoukyaKohakuShizuku() {
    phaseMaxLife = 9500;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    ObjEnemy_SetTimer(enemy, 0);
    phaseActive = true;
    hpPercent = 100;

    Phase3_RainTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
}

task Phase3_RainTask() {
    let wavePos = 0;
    let waveDir = 1;
    let diagToggle = 1;
    while (phaseActive) {
        let diagonal = hpPercent <= 70;
        let n = 5;
        if (hpPercent <= 10) { n = 7; }
        ascent(i in 0..n) {
            let x = 20 + wavePos + i * (340.0 / n);
            if (x < 10) { x = x + 340; }
            if (x > 370) { x = x - 340; }
            let isBig = (rand(0, 1) < 0.15);
            let angle = 90;
            if (diagonal) {
                if (diagToggle > 0) { angle = 65; } else { angle = 115; }
            }
            let sh = CreateShotA1(x, 10, 1, angle, ID_TEAR, 60);
            if (isBig) { ObjRender_SetColor(sh, 200, 140, 60); }
            else { ObjRender_SetColor(sh, 150, 90, 60); }
            Phase3_DropTask(sh, isBig);
        }
        wavePos = wavePos + waveDir * 18;
        if (wavePos > 170 || wavePos < -170) { waveDir = -waveDir; }
        if (diagonal) { diagToggle = -diagToggle; }
        Wait(hpPercent <= 10 ? 22 : 34);
    }
}

task Phase3_DropTask(sh, isBig) {
    Wait(60); // ディレイ中は薄い光の線として表示され、無害
    if (Obj_IsDeleted(sh)) { return; }
    if (isBig) { ObjMove_SetSpeed(sh, 1.3); } else { ObjMove_SetSpeed(sh, 2.2); }
    Wait(50);
    if (Obj_IsDeleted(sh)) { return; }
    // 画面中央付近で加速
    if (isBig) { ObjMove_SetSpeed(sh, 2.6); } else { ObjMove_SetSpeed(sh, 4.2); }

    if (isBig) {
        Wait(20);
        if (!Obj_IsDeleted(sh)) {
            let sx = ObjMove_GetX(sh);
            let sy = ObjMove_GetY(sh);
            ascent(i in 0..6) {
                let a = 60 + i * 60;
                let p = CreateShotA1(sx, sy, 1.4, a, ID_BALL, 1);
                ObjRender_SetColor(p, 150, 90, 60);
                Phase3_FadeParticle(p);
            }
        }
    }

    if (hpPercent <= 40) {
        Wait(80);
        if (!Obj_IsDeleted(sh)) {
            let sx = ObjMove_GetX(sh);
            let sy = ObjMove_GetY(sh);
            let r1 = CreateShotA1(sx, sy, 0.8, 0, ID_BALL, 10);
            ObjRender_SetColor(r1, 220, 200, 170);
            let r2 = CreateShotA1(sx, sy, 0.8, 180, ID_BALL, 10);
            ObjRender_SetColor(r2, 220, 200, 170);
        }
    }
}

task Phase3_FadeParticle(p) {
    Wait(24);
    if (!Obj_IsDeleted(p)) {
        ObjMove_SetSpeed(p, 0);
        Wait(4);
        if (!Obj_IsDeleted(p)) { Obj_Delete(p); }
    }
}

//==================================================================
// フェーズ4: 烏龍茶・龍の茶煙
//==================================================================
function Phase4_OolongRyuuChaen() {
    phaseMaxLife = 9000;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    ObjEnemy_SetTimer(enemy, 0);
    phaseActive = true;
    hpPercent = 100;

    Phase4_BandTask(-1);
    Phase4_BandTask(1);
    Phase4_DropletTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
}

task Phase4_BandTask(side) {
    let t = 0;
    while (phaseActive) {
        let vortex = hpPercent <= 10;
        let hx = baseX;
        let hy = 60;
        if (!vortex) {
            hx = baseX + side * 120 * cos(t * 0.6) + 40 * sin(t * 2.0);
            hy = 60 + 200 * (0.5 + 0.5 * sin(t * 0.35));
        } else {
            let radius = 130;
            let a = t * side * 3.0;
            hx = baseX + radius * cos(a);
            hy = baseY + 90 + radius * 0.6 * sin(a);
        }

        let sh = CreateShotA1(hx, hy, 1.6, 90 + side * 25, ID_BALL, 1);
        ObjRender_SetColor(sh, 130, 190, 190);
        Phase4_FadeChain(sh, 90);

        if (hpPercent <= 70 && (t % 40 < 1)) {
            let sh2 = CreateShotA1(hx, hy, 1.6, 90 - side * 40, ID_BALL, 1);
            ObjRender_SetColor(sh2, 130, 190, 190);
            Phase4_FadeChain(sh2, 60);
        }

        if (hpPercent <= 40 && (t % 90 < 1)) {
            let big = CreateShotA1(GetPlayerX(), 20, 1.0, 90, ID_BIG, 40);
            ObjRender_SetColor(big, 90, 150, 150);
            Phase4_DragonHead(big);
        }

        t = t + 1;
        Wait(3);
    }
}

task Phase4_FadeChain(sh, life) {
    Wait(life);
    if (!Obj_IsDeleted(sh)) {
        ObjMove_SetSpeed(sh, 0.3);
    }
}

task Phase4_DragonHead(big) {
    Wait(90);
    if (!Obj_IsDeleted(big)) {
        let sx = ObjMove_GetX(big);
        let sy = ObjMove_GetY(big);
        ascent(i in 0..5) {
            let p = CreateShotA1(sx, sy - i * 10, 1.5, 90, ID_BALL, 1);
            ObjRender_SetColor(p, 90, 150, 150);
        }
    }
}

task Phase4_DropletTask() {
    while (phaseActive) {
        let sh = CreateShotA1(baseX + rand(-140, 140), baseY, 3.0, 90, ID_NEEDLE, 8);
        ObjRender_SetColor(sh, 220, 200, 130);
        Wait(18);
    }
}

// フェーズ5: ほうじ茶・焙煎火花
//==================================================================
function Phase5_HoujichaBaisenHibana() {
    phaseMaxLife = 9000;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    ObjEnemy_SetTimer(enemy, 0);
    phaseActive = true;
    hpPercent = 100;

    Phase5_EmberTask();
    Phase5_HeatTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
}

task Phase5_EmberTask() {
    let quadrant = 0;
    let flowAngle = 0;
    while (phaseActive) {
        let bx = ObjEnemy_GetX(enemy);
        let by = ObjEnemy_GetY(enemy);
        let furnace = hpPercent <= 10;

        if (!furnace) {
            let startA = quadrant * 90;
            ascent(i in 0..5) {
                let a = startA + i * 22;
                let sh = CreateShotA1(bx + 70 * cos(a), by + 70 * sin(a), 1.4, a, ID_BALL, 1);
                ObjRender_SetColor(sh, 220, 130, 40);
                Phase5_EmberBehavior(sh);
            }
            quadrant = (quadrant + 1) % 4;
            Wait(50);
        } else {
            flowAngle = flowAngle + 6;
            ascent(i in 0..10) {
                let a = flowAngle + i * 36;
                let sh = CreateShotA1(bx + 70 * cos(a), by + 70 * sin(a), 2.6, a, ID_BALL, 1);
                ObjRender_SetColor(sh, 255, 120, 40);
            }
            Wait(16);
        }
    }
}

task Phase5_EmberBehavior(sh) {
    Wait(20);
    if (Obj_IsDeleted(sh)) { return; }
    ObjMove_SetSpeed(sh, 0.2);
    if (hpPercent <= 70) {
        let sx = ObjMove_GetX(sh);
        let sy = ObjMove_GetY(sh);
        ascent(i in 0..4) {
            let p = CreateShotA1(sx, sy, 0.9, i * 90, ID_BALL, 1);
            ObjRender_SetColor(p, 255, 180, 90);
            Phase5_FadeSlow(p);
        }
    }
    Wait(20);
    if (!Obj_IsDeleted(sh)) {
        ObjMove_SetAngle(sh, 90);
        ObjMove_SetSpeed(sh, 1.0);
        Wait(40);
        if (!Obj_IsDeleted(sh)) {
            let sx = ObjMove_GetX(sh);
            let sy = ObjMove_GetY(sh);
            Obj_Delete(sh);
            let glow = CreateShotA1(sx, sy, 0, 0, ID_BALL, 1);
            ObjRender_SetColor(glow, 200, 40, 20);
            ObjRender_SetAlpha(glow, 150);
            Phase5_Afterglow(glow);
        }
    }
}

task Phase5_FadeSlow(p) {
    Wait(30);
    if (!Obj_IsDeleted(p)) { Obj_Delete(p); }
}

task Phase5_Afterglow(glow) {
    Wait(60);
    if (!Obj_IsDeleted(glow)) { Obj_Delete(glow); }
}

task Phase5_HeatTask() {
    while (phaseActive) {
        if (hpPercent <= 40) {
            let x = rand(60, 320);
            let sh = CreateShotA1(x, 420, 1.4, 270, ID_BALL, 1);
            ObjRender_SetColor(sh, 230, 70, 30);
            ObjRender_SetAlpha(sh, 180);
            Phase5_Sway(sh);
        }
        Wait(30);
    }
}

task Phase5_Sway(sh) {
    let t = 0;
    while (!Obj_IsDeleted(sh) && t < 200) {
        ObjMove_SetAngle(sh, 270 + 20 * sin(t * 5));
        t = t + 1;
        yield;
    }
}

//==================================================================
// フェーズ6: ミルクティー・白霞の螺旋
//==================================================================
function Phase6_MilkTeaHakkaRasen() {
    phaseMaxLife = 9500;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    ObjEnemy_SetTimer(enemy, 0);
    phaseActive = true;
    hpPercent = 100;
    spiralPulse = 1.0;

    Phase6_ArmTask();
    Phase6_PulseTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
}

task Phase6_PulseTask() {
    while (phaseActive) {
        if (hpPercent <= 10) {
            let t = 0;
            while (t < 30 && phaseActive) { spiralPulse = 1.0 - 0.4 * (t / 30.0); t = t + 1; yield; }
            t = 0;
            while (t < 50 && phaseActive) { spiralPulse = 0.6 + 0.9 * (t / 50.0); t = t + 1; yield; }
        } else {
            spiralPulse = 1.0;
            yield;
        }
    }
}

task Phase6_ArmTask() {
    let angs = [0.0, 90.0, 180.0, 270.0];
    let dirs = [1, -1, 1, -1];
    while (phaseActive) {
        let bx = ObjEnemy_GetX(enemy);
        let by = ObjEnemy_GetY(enemy);
        ascent(i in 0..4) {
            let a = angs[i];
            let sh = CreateShotA1(bx, by, 1.8 * spiralPulse, a, ID_BIG, 1);
            ObjRender_SetColor(sh, 255, 250, 240);

            let a2 = a + 45;
            let sub = CreateShotA1(bx, by, 1.4 * spiralPulse, a2, ID_BALL, 1);
            ObjRender_SetColor(sub, 190, 150, 110);
            Phase6_SubBehavior(sub, dirs[i]);

            angs[i] = a + dirs[i] * 5;
        }

        if (hpPercent <= 40 && (ObjEnemy_GetInfo(enemy, INFO_TIMER) % 70 == 0)) {
            ascent(i in 0..6) {
                let pa = i * 60;
                let petal = CreateShotA1(bx, by, 1.0, pa, ID_STAR, 1);
                ObjRender_SetColor(petal, 250, 170, 200);
                let sgn = 1;
                if (i % 2 == 0) { sgn = -1; }
                ObjMove_SetAngularVelocity(petal, sgn * 1.2);
            }
        }
        Wait(6);
    }
}

task Phase6_SubBehavior(sub, dir) {
    if (hpPercent > 70) { return; }
    Wait(50);
    if (Obj_IsDeleted(sub)) { return; }
    ascent(i in 0..8) {
        if (Obj_IsDeleted(sub)) { return; }
        Wait(2);
    }
    if (!Obj_IsDeleted(sub)) {
        ObjRender_SetColor(sub, 255, 250, 245);
        ObjMove_SetAngularVelocity(sub, -dir * 1.5);
    }
}

function Phase7_ChakaiShichisaiSenkyakuBanrai() {

    ObjEnemy_SetLife(enemy, phaseMaxLife);
    ObjEnemy_SetTimer(enemy, 0);
    phaseActive = true;
    hpPercent = 100;
    haloAngleOffset = 0;

    Phase7_CycleTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }

    Wait(6);
}

task Phase7_HaloRotateTask() {
    while (phaseActive) {
        if (hpPercent <= 40) { haloAngleOffset = haloAngleOffset + 0.4; }
        yield;
    }
}

task Phase7_CycleTask() {
    while (phaseActive) {
        if (hpPercent <= 10) {
            Phase7_FinalBarrage();
        } else {
            // 通常: 8秒周期(=480f/2秒毎) / HP20%以下で6秒周期(=360f/1.5秒毎)
            let q = 120;
            if (hpPercent <= 20) { q = 90; }
            Phase7_SubLeaf();
            Wait(q);
            if (!phaseActive) { return; }
            Phase7_SubRain();
            Wait(q);
            if (!phaseActive) { return; }
            Phase7_SubMilk();
            Wait(q);
            if (!phaseActive) { return; }
            Phase7_SubSparkSmoke();
            if (hpPercent <= 70) { Phase7_SubBubbleEdge(); }
            Wait(q);
        }
    }
}

function Phase7_SubLeaf() {
    let bx = ObjEnemy_GetX(enemy);
    let by = ObjEnemy_GetY(enemy);
    ascent(i in 0..10) {
        let a = haloAngleOffset + i * 36;
        let sh = CreateShotA1(bx, by, 1.8, a, ID_RICE, 1);
        ObjRender_SetColor(sh, 170, 230, 90);
    }
}

function Phase7_SubRain() {
    ascent(i in 0..6) {
        let x = 30 + i * 60;
        let sh = CreateShotA1(x, 10, 2.0, 90, ID_TEAR, 40);
        ObjRender_SetColor(sh, 180, 110, 60);
    }
}

function Phase7_SubMilk() {
    let bx = ObjEnemy_GetX(enemy);
    let by = ObjEnemy_GetY(enemy);
    ascent(i in 0..4) {
        let a = haloAngleOffset + i * 90;
        let sh = CreateShotA1(bx, by, 1.6, a, ID_BIG, 1);
        ObjRender_SetColor(sh, 255, 250, 240);
        let sgn = 1;
        if (i % 2 == 0) { sgn = -1; }
        ObjMove_SetAngularVelocity(sh, sgn * 2.0);
    }
}

function Phase7_SubSparkSmoke() {
    let bx = ObjEnemy_GetX(enemy);
    let by = ObjEnemy_GetY(enemy);
    ascent(i in 0..8) {
        let a = haloAngleOffset + i * 45;
        let sh = CreateShotA1(bx + 70 * cos(a), by + 70 * sin(a), 1.6, a, ID_BALL, 1);
        ObjRender_SetColor(sh, 220, 130, 40);
    }
    ascent(i in 0..2) {
        let side = -1;
        if (i == 1) { side = 1; }
        let sh2 = CreateShotA1(bx + side * 100, by, 1.4, 90 + side * 20, ID_BALL, 1);
        ObjRender_SetColor(sh2, 120, 190, 190);
    }
}

function Phase7_SubBubbleEdge() {
    let bx = ObjEnemy_GetX(enemy);
    let by = ObjEnemy_GetY(enemy);
    ascent(i in 0..8) {
        // i==0,4 の方向は常に空けておき、画面対角に大きな切れ目を残す
        if (i != 0 && i != 4) {
            let a = haloAngleOffset + i * 45 + 180;
            let sh = CreateShotA1(bx, by, 1.0, a, ID_RING, 1);
            ObjRender_SetColor(sh, 100, 220, 100);
        }
    }
}

task Phase7_FinalBarrage() {
    ObjEnemy_SetX(enemy, baseX);
    ObjEnemy_SetY(enemy, baseY);
    let t = 0;
    let layers = [70, 105, 140, 175, 210, 245, 280];
    let ids = [ID_RICE, ID_RING, ID_TEAR, ID_BALL, ID_BALL, ID_BIG, ID_STAR];
    let colR = [170, 100, 180, 120, 220, 255, 250];
    let colG = [230, 220, 110, 190, 130, 250, 170];
    let colB = [90, 100, 60, 190, 40, 240, 200];
    while (phaseActive && hpPercent <= 10) {
        let bx = ObjEnemy_GetX(enemy);
        let by = ObjEnemy_GetY(enemy);
        let spd = 0.7;
        if (t > 600) { spd = 0.45; } // 終盤は弾速をわずかに落とす
        ascent(L in 0..7) {
            let ringRadius = layers[L];
            let gap = t * (0.8 + L * 0.1) + L * 40;
            let n = 12;
            ascent(i in 0..n) {
                let a = i * (360.0 / n) + t * 0.3;
                if (abs(AngleDiff(a, gap)) > 14) {
                    let sx = bx + ringRadius * cos(a);
                    let sy = by + ringRadius * sin(a);
                    let sh = CreateShotA1(sx, sy, spd, a, ids[L], 20);
                    ObjRender_SetColor(sh, colR[L], colG[L], colB[L]);
                }
            }
        }
        t = t + 4;
        Wait(30);
    }
}
