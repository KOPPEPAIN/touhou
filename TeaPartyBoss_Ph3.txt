#include "script/default_system/Default_ShotConst.txt"

//==================================================================
// 共通ユーティリティ
//==================================================================
function Wait(f) {
    loop(f) { yield; }
}

function AngleDiff(a, b) {
    let d = a - b;
    while (d > 180) { d = d - 360; }
    while (d < -180) { d = d + 360; }
    return d;
}

//==================================================================
// グローバル状態
// 弾グラフィックは Default_ShotConst.txt で定義済みの DS_BALL_<サイズ>_<色>
// （サイズ: SS/S/M/L、色: RED/ORANGE/YELLOW/GREEN/CYAN/PURPLE/GRAY）のみを使用。
// 米粒・涙型・輪等の専用形状定数は環境差が大きいため、あえて使わず
// 「玉」を基本形状として色とサイズと動きだけで各茶種を演出している。
//==================================================================
let enemy;
let baseX = 192;
let baseY = 120;
let hpPercent = 100;
let phaseMaxLife = 1;
let phaseActive = false;

// フェーズ1
let spiralDir = 1;
let spiralPaused = false;
let leafRingArr = [];

// フェーズ2
let ringGapAngle = 0;
let ringPhaseAngle = 0;

// フェーズ6
let spiralPulse = 1.0;
let phase6Counter = 0;

// フェーズ7
let haloAngleOffset = 0;

//==================================================================
// エントリポイント
//==================================================================
@Initialize {
    enemy = ObjEnemy_Create(OBJ_ENEMY_BOSS);
    ObjEnemy_Regist(enemy);
    ObjMove_SetX(enemy, baseX);
    ObjMove_SetY(enemy, baseY);

    CollisionUpdateTask();
    MainTask();
}

@MainLoop {
    yield;
}

@Finalize {
}

task CollisionUpdateTask() {
    while (!Obj_IsDeleted(enemy)) {
        let bx = ObjMove_GetX(enemy);
        let by = ObjMove_GetY(enemy);
        ObjEnemy_SetIntersectionCircleToShot(enemy, bx, by, 32);
        ObjEnemy_SetIntersectionCircleToPlayer(enemy, bx, by, 48);
        yield;
    }
}

task MainTask() {
    Phase1_SenchaWakaba();
    Phase2_MatchaHouwaSuikan();
    Phase3_KoukyaKohakuShizuku();
    Phase4_OolongRyuuChaen();
    Phase5_HoujichaBaisenHibana();
    Phase6_MilkTeaHakkaRasen();
    Phase7_ChakaiShichisaiSenkyakuBanrai();
    Obj_Delete(enemy);
}

//==================================================================
// フェーズ1: 煎茶・若葉の渦
//==================================================================
function Phase1_SenchaWakaba() {
    phaseMaxLife = 9000;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    phaseActive = true;
    spiralDir = 1;
    spiralPaused = false;
    hpPercent = 100;

    leafRingArr = [];
    ascent(i in 0..12) {
        let sh = CreateShotA1(baseX, baseY, 0, i * 30, DS_BALL_S_GREEN, 1);
        leafRingArr = leafRingArr ~ [sh];
    }

    Phase1_LeafRingTask();
    Phase1_SpiralTask();
    Phase1_FlipTask();
    Phase1_GoldLineTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }

    phaseActive = false;
    Wait(6);
    ascent(i in 0..length(leafRingArr)) {
        if (!Obj_IsDeleted(leafRingArr[i])) { Obj_Delete(leafRingArr[i]); }
    }
}

task Phase1_LeafRingTask() {
    let ang = 0;
    while (phaseActive) {
        let bx = ObjMove_GetX(enemy);
        let by = ObjMove_GetY(enemy);
        ascent(i in 0..length(leafRingArr)) {
            if (!Obj_IsDeleted(leafRingArr[i])) {
                let a = ang + i * 30;
                ObjMove_SetX(leafRingArr[i], bx + 44 * cos(a));
                ObjMove_SetY(leafRingArr[i], by + 44 * sin(a));
            }
        }
        ang = ang + spiralDir * 1.2;
        yield;
    }
}

task Phase1_SpiralTask() {
    let angIn = 0;
    let angOut = 0;
    while (phaseActive) {
        if (!spiralPaused) {
            let bx = ObjMove_GetX(enemy);
            let by = ObjMove_GetY(enemy);
            let leafId = DS_BALL_S_GREEN;
            if (hpPercent <= 70) { leafId = DS_BALL_M_GREEN; } // 70%以下は一回り大きく＝深緑を演出

            CreateShotA1(bx, by, 1.6, angIn, leafId, 1);
            CreateShotA1(bx, by, 2.3, angOut, leafId, 1);

            angIn = angIn + spiralDir * 24;
            angOut = angOut + spiralDir * 33;
        }
        if (hpPercent <= 10) { Wait(8); } else { Wait(14); }
    }
}

task Phase1_FlipTask() {
    while (phaseActive) {
        Wait(180);
        if (phaseActive && hpPercent <= 70) {
            spiralPaused = true;
            ascent(i in 0..12) {
                ObjMove_SetX(enemy, baseX + 5 * sin(i * 60));
                Wait(2);
            }
            ObjMove_SetX(enemy, baseX);
            spiralDir = -spiralDir;
            spiralPaused = false;
        }
    }
}

task Phase1_GoldLineTask() {
    while (phaseActive) {
        if (hpPercent <= 40) {
            CreateShotA1(ObjMove_GetX(enemy), ObjMove_GetY(enemy), 3.2, 90, DS_BALL_S_YELLOW, 20);
        }
        if (hpPercent <= 10) { Wait(26); } else { Wait(46); }
    }
}

//==================================================================
// フェーズ2: 抹茶・泡沫翠環
//==================================================================
function Phase2_MatchaHouwaSuikan() {
    phaseMaxLife = 9500;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    phaseActive = true;
    hpPercent = 100;
    ringGapAngle = 0;
    ringPhaseAngle = 0;

    Phase2_SwayTask();
    Phase2_RingTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
    ObjMove_SetX(enemy, baseX);
}

task Phase2_SwayTask() {
    let t = 0;
    while (phaseActive) {
        ObjMove_SetX(enemy, baseX + 36 * sin(t));
        t = t + 1.0;
        yield;
    }
}

task Phase2_RingTask() {
    while (phaseActive) {
        let bx = ObjMove_GetX(enemy);
        let by = ObjMove_GetY(enemy);
        ascent(i in 0..10) {
            let a = ringPhaseAngle + i * 36;
            if (abs(AngleDiff(a, ringGapAngle)) > 18) {
                Phase2_SpawnBubble(bx, by, a, 210, true);
            }
        }
        ascent(i in 0..14) {
            let a = ringPhaseAngle + i * (360.0 / 14.0) + 12;
            if (abs(AngleDiff(a, ringGapAngle)) > 13) {
                Phase2_SpawnBubble(bx, by, a, 130, false);
            }
        }
        if (hpPercent <= 40) {
            ringGapAngle = ringGapAngle + 1.4;
            ringPhaseAngle = ringPhaseAngle + 0.7;
        }
        if (hpPercent <= 10) { Wait(65); } else { Wait(95); }
    }
}

task Phase2_SpawnBubble(bx, by, angle, radius, isBig) {
    let x = bx + radius * cos(angle);
    let y = by + radius * sin(angle);
    let sh;
    if (isBig) { sh = CreateShotA1(x, y, 0, angle, DS_BALL_M_GREEN, 1); }
    else { sh = CreateShotA1(x, y, 0, angle, DS_BALL_S_GREEN, 1); }

    ascent(i in 0..16) {
        if (Obj_IsDeleted(sh)) { return; }
        ObjRender_SetAlpha(sh, 90 + i * 10);
        Wait(2);
    }
    if (isBig) { Wait(30); } else { Wait(14); }
    if (Obj_IsDeleted(sh)) { return; }

    let ang = GetAngleToPlayer(sh);
    ObjMove_SetAngle(sh, ang);
    let spd = 1.9;
    if (isBig) { spd = 1.1; }
    ObjMove_SetSpeed(sh, spd);

    let curveSign = 1;
    if (ObjMove_GetX(sh) < bx) { curveSign = -1; }
    let av = curveSign * 0.28;
    ObjMove_SetAngularVelocity(sh, av);

    if (isBig && hpPercent <= 70) {
        Wait(50);
        if (Obj_IsDeleted(sh)) { return; }
        // 白く点滅して分裂を予告
        let sx = ObjMove_GetX(sh);
        let sy = ObjMove_GetY(sh);
        let a2 = ObjMove_GetAngle(sh);
        let s2 = ObjMove_GetSpeed(sh);
        Obj_Delete(sh);
        let warn = CreateShotA1(sx, sy, s2, a2, DS_BALL_M_GRAY, 1);
        ObjMove_SetAngularVelocity(warn, av);
        Wait(20);
        if (Obj_IsDeleted(warn)) { return; }
        let wx = ObjMove_GetX(warn);
        let wy = ObjMove_GetY(warn);
        let wa = ObjMove_GetAngle(warn);
        let ws = ObjMove_GetSpeed(warn);
        Obj_Delete(warn);
        CreateShotA1(wx, wy, ws, wa - 18, DS_BALL_S_GREEN, 1);
        CreateShotA1(wx, wy, ws, wa + 18, DS_BALL_S_GREEN, 1);
    }
}

//==================================================================
// フェーズ3: 紅茶・琥珀の雫雨
//==================================================================
function Phase3_KoukyaKohakuShizuku() {
    phaseMaxLife = 9500;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    phaseActive = true;
    hpPercent = 100;

    Phase3_RainTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
}

task Phase3_RainTask() {
    let wavePos = 0;
    let waveDir = 1;
    let diagToggle = 1;
    while (phaseActive) {
        let diagonal = hpPercent <= 70;
        let n = 5;
        if (hpPercent <= 10) { n = 7; }
        ascent(i in 0..n) {
            let x = 20 + wavePos + i * (340.0 / n);
            if (x < 10) { x = x + 340; }
            if (x > 370) { x = x - 340; }
            let isBig = (rand(0, 1) < 0.15);
            let angle = 90;
            if (diagonal) {
                if (diagToggle > 0) { angle = 65; } else { angle = 115; }
            }
            let sh;
            if (isBig) { sh = CreateShotA1(x, 10, 1, angle, DS_BALL_M_ORANGE, 60); }
            else { sh = CreateShotA1(x, 10, 1, angle, DS_BALL_S_ORANGE, 60); }
            Phase3_DropTask(sh, isBig);
        }
        wavePos = wavePos + waveDir * 18;
        if (wavePos > 170 || wavePos < -170) { waveDir = -waveDir; }
        if (diagonal) { diagToggle = -diagToggle; }
        if (hpPercent <= 10) { Wait(22); } else { Wait(34); }
    }
}

task Phase3_DropTask(sh, isBig) {
    Wait(60); // ディレイ中は薄い光の線として表示され、無害
    if (Obj_IsDeleted(sh)) { return; }
    if (isBig) { ObjMove_SetSpeed(sh, 1.3); } else { ObjMove_SetSpeed(sh, 2.2); }
    Wait(50);
    if (Obj_IsDeleted(sh)) { return; }
    // 画面中央付近で加速
    if (isBig) { ObjMove_SetSpeed(sh, 2.6); } else { ObjMove_SetSpeed(sh, 4.2); }

    if (isBig) {
        Wait(20);
        if (!Obj_IsDeleted(sh)) {
            let sx = ObjMove_GetX(sh);
            let sy = ObjMove_GetY(sh);
            ascent(i in 0..6) {
                let a = 60 + i * 60;
                let p = CreateShotA1(sx, sy, 1.4, a, DS_BALL_SS_ORANGE, 1);
                Phase3_FadeParticle(p);
            }
        }
    }

    if (hpPercent <= 40) {
        Wait(80);
        if (!Obj_IsDeleted(sh)) {
            let sx = ObjMove_GetX(sh);
            let sy = ObjMove_GetY(sh);
            CreateShotA1(sx, sy, 0.8, 0, DS_BALL_SS_ORANGE, 10);
            CreateShotA1(sx, sy, 0.8, 180, DS_BALL_SS_ORANGE, 10);
        }
    }
}

task Phase3_FadeParticle(p) {
    Wait(24);
    if (!Obj_IsDeleted(p)) {
        ObjMove_SetSpeed(p, 0);
        Wait(4);
        if (!Obj_IsDeleted(p)) { Obj_Delete(p); }
    }
}

//==================================================================
// フェーズ4: 烏龍茶・龍の茶煙
//==================================================================
function Phase4_OolongRyuuChaen() {
    phaseMaxLife = 9000;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    phaseActive = true;
    hpPercent = 100;

    Phase4_BandTask(-1);
    Phase4_BandTask(1);
    Phase4_DropletTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
}

task Phase4_BandTask(side) {
    let t = 0;
    while (phaseActive) {
        let vortex = hpPercent <= 10;
        let hx = baseX;
        let hy = 60;
        if (!vortex) {
            hx = baseX + side * 120 * cos(t * 0.6) + 40 * sin(t * 2.0);
            hy = 60 + 200 * (0.5 + 0.5 * sin(t * 0.35));
        } else {
            let radius = 130;
            let a = t * side * 3.0;
            hx = baseX + radius * cos(a);
            hy = baseY + 90 + radius * 0.6 * sin(a);
        }

        let sh = CreateShotA1(hx, hy, 1.6, 90 + side * 25, DS_BALL_S_CYAN, 1);
        Phase4_FadeChain(sh, 90);

        if (hpPercent <= 70 && (t % 40 < 1)) {
            let sh2 = CreateShotA1(hx, hy, 1.6, 90 - side * 40, DS_BALL_S_CYAN, 1);
            Phase4_FadeChain(sh2, 60);
        }

        if (hpPercent <= 40 && (t % 90 < 1)) {
            let big = CreateShotA1(GetPlayerX(), 20, 1.0, 90, DS_BALL_L_CYAN, 40);
            Phase4_DragonHead(big);
        }

        t = t + 1;
        Wait(3);
    }
}

task Phase4_FadeChain(sh, life) {
    Wait(life);
    if (!Obj_IsDeleted(sh)) {
        ObjMove_SetSpeed(sh, 0.3);
    }
}

task Phase4_DragonHead(big) {
    Wait(90);
    if (!Obj_IsDeleted(big)) {
        let sx = ObjMove_GetX(big);
        let sy = ObjMove_GetY(big);
        ascent(i in 0..5) {
            CreateShotA1(sx, sy - i * 10, 1.5, 90, DS_BALL_S_CYAN, 1);
        }
    }
}

task Phase4_DropletTask() {
    while (phaseActive) {
        CreateShotA1(baseX + rand(-140, 140), baseY, 3.0, 90, DS_BALL_S_YELLOW, 8);
        Wait(18);
    }
}

// フェーズ5: ほうじ茶・焙煎火花
//==================================================================
function Phase5_HoujichaBaisenHibana() {
    phaseMaxLife = 9000;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    phaseActive = true;
    hpPercent = 100;

    Phase5_EmberTask();
    Phase5_HeatTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
}

task Phase5_EmberTask() {
    let quadrant = 0;
    let flowAngle = 0;
    while (phaseActive) {
        let bx = ObjMove_GetX(enemy);
        let by = ObjMove_GetY(enemy);
        let furnace = hpPercent <= 10;

        if (!furnace) {
            let startA = quadrant * 90;
            ascent(i in 0..5) {
                let a = startA + i * 22;
                let sh = CreateShotA1(bx + 70 * cos(a), by + 70 * sin(a), 1.4, a, DS_BALL_S_ORANGE, 1);
                Phase5_EmberBehavior(sh);
            }
            quadrant = (quadrant + 1) % 4;
            Wait(50);
        } else {
            flowAngle = flowAngle + 6;
            ascent(i in 0..10) {
                let a = flowAngle + i * 36;
                CreateShotA1(bx + 70 * cos(a), by + 70 * sin(a), 2.6, a, DS_BALL_S_RED, 1);
            }
            Wait(16);
        }
    }
}

task Phase5_EmberBehavior(sh) {
    Wait(20);
    if (Obj_IsDeleted(sh)) { return; }
    ObjMove_SetSpeed(sh, 0.2);
    if (hpPercent <= 70) {
        let sx = ObjMove_GetX(sh);
        let sy = ObjMove_GetY(sh);
        ascent(i in 0..4) {
            let p = CreateShotA1(sx, sy, 0.9, i * 90, DS_BALL_SS_ORANGE, 1);
            Phase5_FadeSlow(p);
        }
    }
    Wait(20);
    if (!Obj_IsDeleted(sh)) {
        ObjMove_SetAngle(sh, 90);
        ObjMove_SetSpeed(sh, 1.0);
        Wait(40);
        if (!Obj_IsDeleted(sh)) {
            let sx = ObjMove_GetX(sh);
            let sy = ObjMove_GetY(sh);
            Obj_Delete(sh);
            let glow = CreateShotA1(sx, sy, 0, 0, DS_BALL_S_RED, 1);
            ObjRender_SetAlpha(glow, 150);
            Phase5_Afterglow(glow);
        }
    }
}

task Phase5_FadeSlow(p) {
    Wait(30);
    if (!Obj_IsDeleted(p)) { Obj_Delete(p); }
}

task Phase5_Afterglow(glow) {
    Wait(60);
    if (!Obj_IsDeleted(glow)) { Obj_Delete(glow); }
}

task Phase5_HeatTask() {
    while (phaseActive) {
        if (hpPercent <= 40) {
            let x = rand(60, 320);
            let sh = CreateShotA1(x, 420, 1.4, 270, DS_BALL_M_RED, 1);
            ObjRender_SetAlpha(sh, 180);
            Phase5_Sway(sh);
        }
        Wait(30);
    }
}

task Phase5_Sway(sh) {
    let t = 0;
    while (!Obj_IsDeleted(sh) && t < 200) {
        ObjMove_SetAngle(sh, 270 + 20 * sin(t * 5));
        t = t + 1;
        yield;
    }
}

//==================================================================
// フェーズ6: ミルクティー・白霞の螺旋
//==================================================================
function Phase6_MilkTeaHakkaRasen() {
    phaseMaxLife = 9500;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    phaseActive = true;
    hpPercent = 100;
    spiralPulse = 1.0;
    phase6Counter = 0;

    Phase6_ArmTask();
    Phase6_PulseTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
}

task Phase6_PulseTask() {
    while (phaseActive) {
        if (hpPercent <= 10) {
            let t = 0;
            while (t < 30 && phaseActive) { spiralPulse = 1.0 - 0.4 * (t / 30.0); t = t + 1; yield; }
            t = 0;
            while (t < 50 && phaseActive) { spiralPulse = 0.6 + 0.9 * (t / 50.0); t = t + 1; yield; }
        } else {
            spiralPulse = 1.0;
            yield;
        }
    }
}

task Phase6_ArmTask() {
    let angs = [0.0, 90.0, 180.0, 270.0];
    let dirs = [1, -1, 1, -1];
    while (phaseActive) {
        let bx = ObjMove_GetX(enemy);
        let by = ObjMove_GetY(enemy);
        ascent(i in 0..4) {
            let a = angs[i];
            CreateShotA1(bx, by, 1.8 * spiralPulse, a, DS_BALL_L_GRAY, 1);

            let a2 = a + 45;
            let sub = CreateShotA1(bx, by, 1.4 * spiralPulse, a2, DS_BALL_S_ORANGE, 1);
            Phase6_SubBehavior(sub, dirs[i]);

            angs[i] = a + dirs[i] * 5;
        }

        phase6Counter = phase6Counter + 1;
        if (hpPercent <= 40 && (phase6Counter % 12 == 0)) {
            ascent(i in 0..6) {
                let pa = i * 60;
                let petal = CreateShotA1(bx, by, 1.0, pa, DS_BALL_S_PURPLE, 1);
                let sgn = 1;
                if (i % 2 == 0) { sgn = -1; }
                ObjMove_SetAngularVelocity(petal, sgn * 1.2);
            }
        }
        Wait(6);
    }
}

task Phase6_SubBehavior(sub, dir) {
    if (hpPercent > 70) { return; }
    Wait(50);
    if (Obj_IsDeleted(sub)) { return; }
    let sx = ObjMove_GetX(sub);
    let sy = ObjMove_GetY(sub);
    let a = ObjMove_GetAngle(sub);
    let s = ObjMove_GetSpeed(sub);
    Obj_Delete(sub);
    let sh2 = CreateShotA1(sx, sy, s, a, DS_BALL_S_GRAY, 1);
    ObjMove_SetAngularVelocity(sh2, -dir * 1.5);
}

//==================================================================
// フェーズ7: 茶会・七彩千客万来（最終フェーズ）
//==================================================================
function Phase7_ChakaiShichisaiSenkyakuBanrai() {
    phaseMaxLife = 12000;
    ObjEnemy_SetLife(enemy, phaseMaxLife);
    phaseActive = true;
    hpPercent = 100;
    haloAngleOffset = 0;

    Phase7_CycleTask();
    Phase7_HaloRotateTask();

    while (ObjEnemy_GetInfo(enemy, INFO_LIFE) > 0) {
        hpPercent = ObjEnemy_GetInfo(enemy, INFO_LIFE) / phaseMaxLife * 100;
        yield;
    }
    phaseActive = false;
    Wait(6);
}

task Phase7_HaloRotateTask() {
    while (phaseActive) {
        if (hpPercent <= 40) { haloAngleOffset = haloAngleOffset + 0.4; }
        yield;
    }
}

task Phase7_CycleTask() {
    while (phaseActive) {
        if (hpPercent <= 10) {
            Phase7_FinalBarrage();
        } else {
            // 通常: 8秒周期(=480f/2秒毎) / HP20%以下で6秒周期(=360f/1.5秒毎)
            let q = 120;
            if (hpPercent <= 20) { q = 90; }
            Phase7_SubLeaf();
            Wait(q);
            if (!phaseActive) { return; }
            Phase7_SubRain();
            Wait(q);
            if (!phaseActive) { return; }
            Phase7_SubMilk();
            Wait(q);
            if (!phaseActive) { return; }
            Phase7_SubSparkSmoke();
            if (hpPercent <= 70) { Phase7_SubBubbleEdge(); }
            Wait(q);
        }
    }
}

function Phase7_SubLeaf() {
    let bx = ObjMove_GetX(enemy);
    let by = ObjMove_GetY(enemy);
    ascent(i in 0..10) {
        let a = haloAngleOffset + i * 36;
        CreateShotA1(bx, by, 1.8, a, DS_BALL_S_GREEN, 1);
    }
}

function Phase7_SubRain() {
    ascent(i in 0..6) {
        let x = 30 + i * 60;
        CreateShotA1(x, 10, 2.0, 90, DS_BALL_S_ORANGE, 40);
    }
}

function Phase7_SubMilk() {
    let bx = ObjMove_GetX(enemy);
    let by = ObjMove_GetY(enemy);

function Phase7_SubSparkSmoke() {

    }

function Phase7_SubBubbleEdge() {

        if (i != 0 && i != 4) {

    }
}

task Phase7_FinalBarrage() {
    ObjMove_SetX(enemy, baseX);
    ObjMove_SetY(enemy, baseY);
    let t = 0;
    let layers = [70, 105, 140, 175, 210, 245, 280];
    let ids = [DS_BALL_S_GREEN, DS_BALL_S_GRAY, DS_BALL_S_ORANGE, DS_BALL_S_CYAN, DS_BALL_S_RED, DS_BALL_S_GRAY, DS_BALL_S_PURPLE];
    while (phaseActive && hpPercent <= 10) {
        let bx = ObjMove_GetX(enemy);
        let by = ObjMove_GetY(enemy);
        let spd = 0.7;
        if (t > 600) { spd = 0.45; } // 終盤は弾速をわずかに落とす
        ascent(L in 0..7) {
            let ringRadius = layers[L];
            let gap = t * (0.8 + L * 0.1) + L * 40;
            let n = 12;
            ascent(i in 0..n) {
                let a = i * (360.0 / n) + t * 0.3;
                if (abs(AngleDiff(a, gap)) > 14) {
                    let sx = bx + ringRadius * cos(a);
                    let sy = by + ringRadius * sin(a);
                    CreateShotA1(sx, sy, spd, a, ids[L], 20);
                }
            }
        }
        t = t + 4;
        Wait(30);
    }
}
